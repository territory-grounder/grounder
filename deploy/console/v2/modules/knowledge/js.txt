/* ============================================================================
   KNOWLEDGE module (know-) — the living wiki.
   The predecessor accumulated knowledge into a hand-maintained 941-page MkDocs
   site. TG's knowledge is DERIVED instead of authored: every page is composed
   from the estate graph (what an entity depends on), the incidents that touched
   it, and the verifier's verdicts (what was learned). Nothing here is typed by
   hand — so it can never go stale relative to the running estate.
   Two panes: a searchable index of pages (hosts · incident precedents ·
   operational rules) and the selected page. Epistemics preserved: observed
   dependencies render TERRITORY-steel, learned/predicted notes render MAP-indigo.
   ============================================================================ */

let knowSel = null;   // {kind, id}
let knowQuery = "";

function knowSec(lbl, title, ...extra) {
  return h("div", { class: "sec" }, h("span", { class: "lbl" }, lbl), h("h3", {}, title),
    h("span", { class: "rule" }), ...extra);
}
function knowVerdictTag(v) {
  const cls = v === "match" ? "match" : v === "partial" ? "partial" : v === "deviation" ? "deviation" : "";
  return h("span", { class: "know-verdict " + cls }, v || "—");
}
function knowBandTag(b) {
  const cls = b === "AUTO" ? "auto" : b === "AUTO_NOTICE" ? "notice" : "pause";
  return h("span", { class: "bandtag " + cls }, b || "—");
}

/* build the flat page list from the knowledge model */
function knowPages(model) {
  const pages = [];
  (model.hosts || []).forEach(host => pages.push({ kind: "host", id: host.name, title: host.name, sub: host.role, data: host }));
  (model.incidents || []).forEach(inc => pages.push({ kind: "incident", id: inc.ref, title: inc.ref, sub: (inc.host || "") + " · " + (inc.verdict || ""), data: inc }));
  // ★ AUTHORED, NOT DERIVED. These carry authored:true from the model (see modules/_live) and are marked
  // here so the count, the badge and the group heading can all tell the truth about them.
  (model.rules || []).forEach((r, i) => pages.push({ kind: "rule", id: "rule-" + i, title: r.title,
    sub: r.authored ? "authored doctrine — not derived" : "operational rule", data: r, authored: !!r.authored }));
  return pages;
}

/* ---- page renderers ---- */
function knowHostPage(host) {
  const pane = h("div");
  pane.append(knowSec("host · derived from the estate graph", host.name,
    h("span", { class: "know-role" }, host.role || host.type)));
  const depTable = (rows, dir) => {
    if (!rows || !rows.length) return h("p", { class: "know-empty-line" }, dir === "out" ? "No recorded dependencies." : "Nothing depends on this entity yet.");
    const t = h("table", { class: "tbl know-tbl" });
    t.append(h("thead", {}, h("tr", {}, ...[dir === "out" ? "Depends on" : "Depended on by", "Relation", "Conf", "Source"].map(x => h("th", {}, x)))));
    const tb = h("tbody");
    rows.forEach(e => tb.append(h("tr", {},
      h("td", { class: "m", style: "color:var(--territory)" }, e.to || e.from),
      h("td", { class: "m" }, e.rel),
      h("td", { class: "m", style: "color:var(--ink3)" }, (e.confidence != null ? e.confidence.toFixed(2) : "—")),
      h("td", { class: "m", style: "color:var(--ink4)" }, e.source || "—"))));
    t.append(tb); return t;
  };
  pane.append(h("div", { class: "know-block" }, h("div", { class: "know-h" }, "Dependencies"), depTable(host.depends_on, "out")));
  pane.append(h("div", { class: "know-block" }, h("div", { class: "know-h" }, "Dependents"), depTable(host.dependents, "in")));
  /* THREE outcomes here, not two. "a quiet host" is a CLAIM about the estate and it was printed whenever
     the list was empty — including when the sessions read had failed (null, coalesced away upstream) and
     including when the host simply had nothing inside the bounded recent window. It was also, until the
     host field reached this DTO, the answer for EVERY host on every load. A window that cannot see an
     incident is not evidence there was none. */
  /* THE WINDOW'S APOLOGY IS ONLY HONEST WHILE NOTHING BETTER EXISTS. This pane filters a 200-row
     ESTATE-WIDE session window per host, so for 78 hosts it can only ever show a slice — which is why it
     apologised. The wiki compiler now produces a `host-<name>` page from a PER-HOST read with no shared
     window, so when one exists the honest thing is to point at it rather than apologise.

     THREE states, not two, and the third is the one this codebase keeps getting wrong. `null` means the
     wiki index READ FAILED; an index whose articles list is empty means it WAS read and nothing is
     compiled. Collapsing them makes an unreachable wiki render exactly like one that has never run.
     An earlier version of this block did collapse them while this comment claimed it did not — a
     property asserted in prose and enforced nowhere, and the mutation restoring the collapse passed
     cleanly because nothing could tell the difference. */
  const live = typeof liveState !== "undefined" && liveState.on;
  const wikiUnknown = live && liveState.wiki == null;
  const compiledPage = (function () {
    const idx = live ? liveState.wiki : null;
    if (!idx || !Array.isArray(idx.articles)) return null;
    // host.NAME, not host.id — this pane reads host.name everywhere else (knowSec above, and the page
    // index that produced it). Reading a field the object does not carry yields "", the lookup never
    // matches, and the feature is silently off: the exact shape of the hostOf() defect fixed in !824,
    // reproduced in the code that fixes its sibling.
    const want = "host-" + String(host.name || "").toLowerCase();
    return idx.articles.some(a => a && a.slug === want) ? want : null;
  })();
  const sessionsUnknown = typeof liveState !== "undefined" && liveState.on && liveState.sessions == null;
  const windowed = typeof liveState !== "undefined" && liveState.on
    && Array.isArray(liveState.sessions) && liveState.sessions.length >= 200;
  const compiledNote = compiledPage
    ? h("p", { class: "know-empty-line" },
        "This list is the newest 200 sessions across the estate, filtered to this host. ",
        h("a", { class: "know-link m", href: "#wiki", onclick: e => { e.preventDefault(); knowOpenCompiled(compiledPage); } },
          "The compiled page"),
        " has the complete per-host record.")
    : wikiUnknown
      ? h("p", { class: "know-empty-line" },
          "Drawn from the newest 200 sessions across the whole estate — this host may have older ones. " +
          "The wiki index could not be read, so whether a complete compiled page exists for this host is unknown.")
      : (windowed ? h("p", { class: "know-empty-line" },
          "Drawn from the newest 200 sessions across the whole estate — this host may have older ones.") : null);
  pane.append(h("div", { class: "know-block" }, h("div", { class: "know-h" }, "Incidents seen here"),
    (host.incidents && host.incidents.length)
      ? h("div", { class: "know-inc-list" },
          ...host.incidents.map(inc => h("div", { class: "know-inc" },
            h("a", { class: "know-link m", href: "#knowledge", onclick: e => { e.preventDefault(); knowGo("incident", inc.ref); } }, inc.ref),
            knowBandTag(inc.band), knowVerdictTag(inc.verdict),
            h("span", { class: "know-inc-note" }, inc.note || ""))),
          compiledNote)
      : sessionsUnknown
        ? h("p", { class: "know-empty-line" }, "The session read failed — this host's incidents could not be read, which is not the same as it having none.")
        : compiledPage
          ? h("p", { class: "know-empty-line" },
              "None in the newest 200 sessions across the estate. ",
              h("a", { class: "know-link m", href: "#wiki", onclick: e => { e.preventDefault(); knowOpenCompiled(compiledPage); } },
                "The compiled page for this host"),
              " is built from a per-host read with no shared window — it is the whole record.")
          : windowed
            ? h("p", { class: "know-empty-line" }, "None in the newest 200 sessions across the estate. That is a window, not the whole record — this host may have older incidents.")
            : h("p", { class: "know-empty-line" }, "No incidents recorded against this entity — a quiet host.")));
  return pane;
}
/* Open a compiled article in the wiki view. The slug is derived from the host name and validated against
   the served index above, so nothing here trusts a value to be a real page. */
function knowOpenCompiled(slug) {
  if (typeof wkUI === "object") { wkUI.sel = slug; }
  if (typeof facetState === "object") { facetState.wiki = "articles"; }
  location.hash = "wiki";
  if (typeof route === "function") route("wiki");
}

function knowIncidentPage(inc) {
  const pane = h("div");
  pane.append(knowSec("incident precedent · derived from the audit spine", inc.ref,
    knowBandTag(inc.band), knowVerdictTag(inc.verdict)));
  pane.append(h("div", { class: "know-meta" },
    h("div", {}, h("span", { class: "know-k" }, "Entity"), h("a", { class: "know-link m", href: "#knowledge", onclick: e => { e.preventDefault(); knowGo("host", inc.host); } }, inc.host || "—")),
    h("div", {}, h("span", { class: "know-k" }, "Risk"), h("span", { class: "m" }, inc.risk || "—")),
    h("div", {}, h("span", { class: "know-k" }, "Classified"), h("span", { class: "m", style: "color:var(--ink3)" }, inc.classified ? String(inc.classified).slice(0, 19).replace("T", " ") : "—"))));
  pane.append(h("div", { class: "know-block" }, h("div", { class: "know-h" }, "What was learned"),
    h("p", { class: "know-learned" }, inc.learned || "No precedent note recorded for this incident.")));
  return pane;
}
function knowRulePage(rule) {
  const pane = h("div");
  pane.append(knowSec("operational rule · invariant of the running system", rule.title));
  pane.append(h("p", { class: "know-learned" }, rule.body));
  return pane;
}

function knowRenderPage(page) {
  if (!page) return h("div", { class: "empty", style: "padding:40px 20px;text-align:center" },
    h("div", { class: "big" }, "Select a page"),
    h("div", { class: "desc" }, "Pick a host, an incident precedent, or an operational rule from the index."));
  if (page.kind === "host") return knowHostPage(page.data);
  if (page.kind === "incident") return knowIncidentPage(page.data);
  return knowRulePage(page.data);
}

/* navigate to a page and refresh the surface in place */
function knowGo(kind, id) {
  knowSel = { kind, id };
  const host = document.querySelector(".main .view");
  if (host) { host.innerHTML = ""; host.append(views.knowledge()); }
}

/* the surface: an index rail + the selected page */
function knowRender(model, live) {
  const pages = knowPages(model);
  const wrap = h("div", { class: "know-wrap" });

  wrap.append(h("div", { class: "know-lead" },
    h("div", { class: "lbl", style: "color:" + (live ? "var(--match)" : "var(--ink3)") },
      live ? "LIVE · derived knowledge" : "design preview · derived knowledge"),
    h("h2", { class: "know-lead-title" }, "A wiki the estate writes itself"),
    h("p", { class: "know-lead-desc" },
      "Every page below is composed from the estate graph, the incidents that touched an entity, and the ",
      "verifier's verdicts — never hand-authored, so it cannot drift from the running system. ",
      // ★ THE COUNT MUST MATCH THE CLAIM. The sentence above says every page is composed and never
      // hand-authored; counting the authored doctrine pages into that total made the number a witness to a
      // false statement. Derived pages are counted here; authored ones are named separately below.
      (function(){
        const derived = pages.filter(p => !p.authored).length;
        const authored = pages.length - derived;
        if(!pages.length) return "No pages yet.";
        return h("span", {}, h("b", {}, String(derived)), " derived page" + (derived === 1 ? "" : "s") +
          (authored ? " · " + authored + " authored doctrine page" + (authored === 1 ? "" : "s") +
            " (hand-written, shown separately — they can drift from the code and are NOT verified against it)" : "."));
      })(),
      // STATE THE COVERAGE. These pages are composed from the sessions read at limit=50; the spine holds
      // far more. Presenting a wiki built from the most recent slice as though it covered the whole
      // incident history is the same defect as a badge reporting its fetch limit — it reads as complete.
      // Rendered only when the control plane supplied a real total, so an older one says nothing rather
      // than something wrong.
      (typeof liveState !== "undefined" && liveState.on && liveState.sessionTotal &&
       Array.isArray(liveState.sessions) && liveState.sessionTotal > liveState.sessions.length)
        ? h("span", { class: "lbl", style: "display:block;margin-top:6px;text-transform:none;color:var(--ink4)" },
            "Composed from the most recent " + liveState.sessions.length + " of " +
            liveState.sessionTotal + " recorded incidents — older ones are on the spine, not on this page.")
        : null)));

  const grid = h("div", { class: "know-grid" });

  // ---- index rail (searchable, grouped) ----
  const rail = h("div", { class: "know-index" });
  const search = h("input", { class: "know-search m", type: "search", "aria-label": "Filter knowledge pages", placeholder: "filter pages…", value: knowQuery,
    oninput: e => { knowQuery = e.target.value; const host = document.querySelector(".main .view"); if (host) { host.innerHTML = ""; host.append(views.knowledge()); const s = host.querySelector(".know-search"); if (s) { s.focus(); s.setSelectionRange(s.value.length, s.value.length); } } } });
  rail.append(search);
  const q = knowQuery.trim().toLowerCase();
  const groups = [["Hosts", "host"], ["Incident precedents", "incident"], ["Operational rules — AUTHORED, not derived", "rule"]];
  let shown = 0;
  groups.forEach(([label, kind]) => {
    const items = pages.filter(p => p.kind === kind && (!q || (p.title + " " + (p.sub || "")).toLowerCase().includes(q)));
    if (!items.length) return;
    rail.append(h("div", { class: "know-grp" }, label));
    items.forEach(p => {
      shown++;
      const active = knowSel && knowSel.kind === p.kind && knowSel.id === p.id;
      rail.append(h("a", { class: "know-item" + (active ? " active" : ""), href: "#knowledge",
        onclick: e => { e.preventDefault(); knowGo(p.kind, p.id); } },
        h("span", { class: "know-item-t m" }, p.title),
        h("span", { class: "know-item-s" }, p.sub || "")));
    });
  });
  if (!shown) rail.append(h("div", { class: "know-empty-line", style: "padding:14px 10px" }, "No pages match “" + knowQuery + "”."));
  grid.append(rail);

  // ---- selected page ----
  let sel = null;
  if (knowSel) sel = pages.find(p => p.kind === knowSel.kind && p.id === knowSel.id) || null;
  if (!sel && pages.length) { sel = pages[0]; knowSel = { kind: sel.kind, id: sel.id }; }
  grid.append(h("div", { class: "know-page wrapcard" }, knowRenderPage(sel)));

  wrap.append(grid);
  return wrap;
}

function knowView() {
  return knowRender(typeof KNOW !== "undefined" ? KNOW : { hosts: [], incidents: [], rules: [] }, false);
}
