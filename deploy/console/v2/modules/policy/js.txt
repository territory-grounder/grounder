/* relative time from an RFC3339 string; honest "never"/passthrough on empty/unparseable. */
function polRel(iso) {
  if (!iso) return "never";
  const t = Date.parse(iso);
  if (isNaN(t)) return String(iso);
  let s = Math.floor((Date.now() - t) / 1000);
  if (s < 0) return "just now";
  if (s < 60) return s + "s ago";
  const m = Math.floor(s / 60); if (m < 60) return m + "m ago";
  const hh = Math.floor(m / 60); if (hh < 24) return hh + "h ago";
  const d = Math.floor(hh / 24); if (d < 30) return d + "d ago";
  const mo = Math.floor(d / 30); if (mo < 12) return mo + "mo ago";
  return Math.floor(mo / 12) + "y ago";
}

/* verdict pill — auto (map-indigo) / approve (amber) / deny (deviation), rules-as-data semantics. */

  const t = String(v || "").toLowerCase();
  const tint = t === "auto" ? "var(--map)" : t === "approve" ? "var(--partial)" : t === "deny" ? "var(--deviation)" : "var(--ink3)";

/* graduation level pill — auto = earned autonomy (map), approve = human-gated (steel). */

/* ---- TG-106: the autonomy-mode WRITE control — a gated, double-warned, admin-elevated transition ----------
   The grounder NEVER flips the mode itself (spec/015 REQ-1502): this UI composes the operator's order and
   POSTs {to, expected_from, reason} to /api/v1/mode, where the WORKER's chokepoint-bound ModeController is the
   LAST gate (the wired AuthorityChecker + the green boot preflight decide, and every attempt is audited to the
   hash chain). It is honest by construction — real states only: idle (showing the live mode) · choosing a
   target · a RED double-warn for the allow-all posture OR a plain confirm · submitting · the honest committed
   ModeTransitionOutcome · an honest refusal. It NEVER fabricates a mode or a success (INV-15). It requires an
   admin-ELEVATED session (the SAME step-up config/secret writes use, reused verbatim): a 401 drives that
   step-up, a 403 is an honest authZ refusal (elevation cannot fix "not flip-authorized"/cross-origin), a 409
   says the mode changed underneath us (re-read, never silently overwrite), a 503/network is honest. It sends
   ONLY {to, expected_from, reason} + the same-origin elevated cookie — no secret, no bypass of the backend's
   adminAuthorized/chokepoint gate. */
const POL_MODES = ["Shadow", "HITL", "Semi-auto", "Full-auto"];

/* Canonical spelling of a fetched mode (case-insensitive). The worker's ParseMode is case-SENSITIVE, so the
   expected_from CAS guard must carry the exact wire spelling; an unrecognised value → null (no live mode). */
function polCanonMode(s) { return POL_MODES.find(m => m.toLowerCase() === String(s || "").toLowerCase()) || null; }

/* The RED double-warn trigger (TG-106's defining safety rule). Full-auto is the ONLY allow-all posture among
   the four modes — core/policy/mode.go ModeFullAuto "applies the auto-approve preset: any allow/auto verdict
   actuates except the constitutional never-auto floor", i.e. it REMOVES the operator deny-floor's protections.
   Shadow/HITL never auto-actuate; Semi-auto still gates non-graduated classes to a vote. Entering Full-auto is
   the transition that throws the RED confirm; every other target uses a single plain confirm. */
function polModeIsAllowAll(to) { return String(to || "").toLowerCase() === "full-auto"; }

/* The live current mode = the CAS expected_from. undefined/null policyMode, or an unknown mode name, → null,
   and the selector then stays read-only: a compare-and-swap needs a known baseline to swap against. */
function polModeCurrent() { const m = liveState.policyMode; return (m && m.mode) ? polCanonMode(m.mode) : null; }

const polMode = { step: "select", target: null, reason: "", busy: false, outcome: null, error: null, stale: false };

function polModeReset() { polMode.step = "select"; polMode.target = null; polMode.reason = ""; polMode.busy = false; polMode.outcome = null; polMode.error = null; polMode.stale = false; }

/* Map a mode-POST failure to an honest sentence + flags. Grounded in the REAL contract (core/httpapi
   mode_write.go + core/auth AuthAdminSession): 401 = absent/expired/plain-session → drive the step-up; 403 =
   cross-origin OR not-in-the-flip-authorized-set → an honest terminal refusal (elevation cannot fix either);
   409 = stale expected_from ("changed underneath you") OR the boot preflight is not green; 400 = malformed /
   bad mode / missing reason; 429 rate-limited; 503/404/405 the worker did not accept the flip (fail-closed). */
async function polModePostFail(r) {
  let server = ""; try { server = (await r.text()).trim().slice(0, 300); } catch (e) {}
  if (r.status === 401) return { elevate: true, msg: "admin elevation required — step up to change the autonomy mode" };
  if (r.status === 403) return { msg: "refused: " + (server || "this session may not change the autonomy mode (not in the flip-authorized set, or a cross-origin write)") };
  if (r.status === 409) {
    if (/preflight/i.test(server)) return { msg: "refused: " + (server || "cannot escalate — the boot preflight is not green; the mode was NOT changed") };
    return { stale: true, msg: "the mode changed underneath you — " + (server || "expected_from no longer matches the active mode") + ". Re-read the current mode and retry; nothing was overwritten." };
  }
  if (r.status === 400) return { msg: "refused: " + (server || "malformed mode transition — a target mode and a rationale are required") };
  if (r.status === 429) return { msg: "rate limited — wait a moment and retry" };
  if (r.status === 503 || r.status === 404 || r.status === 405) return { msg: "the mode-transition path is unavailable — the worker did not accept the flip (fail-closed); the mode was NOT changed" };
  return { msg: "mode change failed (" + r.status + ")" + (server ? " — " + server : "") };
}

/* Re-open the dialog straight at the confirm the operator already saw (used after the elevation step-up):
   the pending target+reason survive on module state, so ONE Proceed completes the already-warned change and
   nothing auto-submits. Falls back to the select step if the pending order is incomplete. */
function polOpenModeAtConfirm() {
  const target = polMode.target, reason = polMode.reason;
  polOpenModeDialog();
  if (!document.getElementById("polModeDlg")) return; // no live mode → cannot proceed
  polMode.target = target; polMode.reason = reason;
  const c = polModeCurrent();
  polMode.step = (c && target && target !== c && reason && reason.trim()) ? "confirm" : "select";
  polModeRender();
}

function polModeRender() {
  const body = document.getElementById("polModeBody");
  if (!body) return;
  body.innerHTML = "";
  if (polMode.step === "confirm") body.append(polModeConfirmStep());
  else if (polMode.step === "submitting") body.append(polModeBusyStep());
  else if (polMode.step === "result") body.append(polModeResultStep());
  else if (polMode.step === "error") body.append(polModeErrorStep());
  else body.append(polModeSelectStep());
}

function polModeSelectStep() {
  const c = polModeCurrent();
  const wrap = h("div", { style: "padding:8px 20px 20px;display:flex;flex-direction:column;gap:13px" });
  wrap.append(h("p", { class: "pol-note" },
    "The active mode is ", h("b", {}, c),
    ". Choose a target mode and state why — the change is recorded to the governance ledger and executed by the worker; this console never flips the mode itself."));

  const sel = h("select", { class: "pol-tin", id: "polModeTarget", "aria-label": "target mode",
    onchange: e => { polMode.target = e.target.value; sync(); } });
  POL_MODES.forEach(mm => sel.append(h("option", { value: mm }, mm + (mm === c ? "  (current)" : ""))));
  const start = polMode.target || c;
  sel.value = start; polMode.target = start;

  const reason = h("textarea", { class: "pol-mreason", id: "polModeReason", "aria-label": "reason",
    placeholder: "reason — required; recorded to the governance ledger before the flip",
    oninput: e => { polMode.reason = e.target.value; sync(); } });
  reason.value = polMode.reason || "";

  const warnBox = h("div", { id: "polModeWarnBox" });
  const go = h("button", { class: "btn primary", type: "button", id: "polModeReview", disabled: "", onclick: () => polModeToConfirm() }, "Review change");
  const cancel = h("button", { class: "btn", type: "button", onclick: () => polModeClose() }, "Cancel");

  function sync() {
    const changed = polMode.target && polMode.target !== c;
    go.disabled = !(changed && polMode.reason.trim());
    warnBox.innerHTML = "";
    if (changed && polModeIsAllowAll(polMode.target)) {
      warnBox.append(h("div", { class: "pol-mwarn-red", role: "alert" },
        h("div", { class: "pol-mwarn-title" }, "Full-auto — allow-all actuation"),
        h("p", { style: "margin:5px 0 0" },
          "Switching into Full-auto authorizes ", h("b", {}, "autonomous, potentially irreversible action across your estate"),
          " — any allow/auto verdict actuates except the constitutional never-auto floor. You will be asked to confirm this in RED.")));
    }
  }

  wrap.append(
    h("label", { class: "pol-tfield" }, h("span", { class: "pol-tlbl" }, "Target mode"), sel),
    h("label", { class: "pol-tfield" }, h("span", { class: "pol-tlbl" }, "Reason ", h("span", { style: "color:var(--deviation)" }, "*")), reason),
    warnBox,
    h("div", { class: "pol-trow" }, go, cancel));
  sync();
  return wrap;
}

function polModeBusyStep() {
  return h("div", { class: "empty", style: "padding:24px 20px" }, h("div", { class: "desc" }, "Submitting the mode transition to the worker…"));
}

async function polModeSubmit() {
  const c = polModeCurrent();
  // The POST is reachable ONLY from the confirm step's Proceed button, so the double-warn has already run; this
  // guard just refuses an incomplete/no-op order (defense in depth — never a mode change with no baseline).
  if (!c || !polMode.target || polMode.target === c || !polMode.reason.trim()) return;
  polMode.step = "submitting"; polMode.busy = true; polMode.error = null; polModeRender();

  const payload = { to: polMode.target, expected_from: c, reason: polMode.reason.trim() };
  let r = null;
  try {
    r = await fetch("/api/v1/mode", { method: "POST", credentials: "same-origin",
      headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) });
  } catch (e) {}
  polMode.busy = false;

  if (r && r.ok) {
    let out = {}; try { out = await r.json(); } catch (e) {}
    polMode.outcome = out; polMode.error = null; polMode.step = "result"; polModeRender();
    await polRefreshMode(); // re-read the live mode so the hero/badge update honestly on close
    return;
  }
  const f = r ? await polModePostFail(r) : { msg: "network error — the control plane did not answer; the mode was NOT changed. Retry." };
  if (f.elevate) {
    // Reuse the SHARED admin step-up (config/secret writes use the same modal + /api/v1/session/elevate). Close
    // this dialog first so only one modal is live, then on a successful step-up re-open straight at the confirm
    // the operator already saw — the double-warn is re-shown and NOTHING auto-submits.
    polModeClose();
    if (typeof cfgElevateOverlay === "function") cfgElevateOverlay("Changing the autonomy mode needs an elevated session.", () => polOpenModeAtConfirm());
    return;
  }
  polMode.error = f.msg; polMode.stale = !!f.stale; polMode.step = "error"; polModeRender();
}

/* A 409 stale is a compare-and-swap loss: re-read the live mode and RESET the order (do not silently reuse the
   old target against a moved baseline). The operator re-composes against the truth — never a silent overwrite. */
async function polModeReread() {
  await polRefreshMode();

/* ---- TG-104 slice-2: the RULES-AS-DATA EDITOR — a gated, ledgered, admin-elevated compare-and-swap write ----
   The grounder NEVER writes the ruleset itself (spec/015 REQ-1503): this UI reads the live rules, lets an admin
   edit them as ordered per-rule cards, composes the FULL replacement document, and POSTs {document,
   expected_version, rationale} to /api/v1/policy/ruleset, where the WORKER validates (core/policy.ParseRuleSet
   is the authority — a malformed ruleset is refused, never persisted), compare-and-swaps against the pinned
   version, ledgers, and applies it. It is honest by construction — real states only: editing (the card list) ·
   saving · the honest committed outcome · an honest refusal · a fail-closed BLOCK for a ruleset it cannot
   represent. It reuses the SAME admin step-up config/secret + the mode change use (a 401 drives it), and it
   NEVER fabricates a success (INV-15).
   THE SAFETY PROPERTY (why this slice exists): the ruleset governs autonomous actuation, so a faithful
   read→edit→write round-trip must drop NOTHING — not a rule's match dimension, not a param, not the order, not
   a top-level default. A dropped deny or an implicit default flip is how a `deny` silently becomes an `allow`.
   Two mechanisms enforce it: (1) the read→write selector-kind map below is a single explicit table, so a
   hyphenated read kind can never be silently mis-bound to the wrong underscore write key or dropped; (2) a
   ruleset carrying a top-level `default` block the read surface can't project (has_default) FAILS CLOSED — the
   editor refuses to open/save rather than emit a document missing that default. Live the shipped ruleset has no
   default, so branch (2) is inert, but the guard exists and is tested. */
const POL_VERDICTS = ["auto", "approve", "deny"];
const POL_BAND_MODES = ["", "respect", "force"];
/* The estate-selector KINDS as the READ renders them — the credential.SelectorKind spellings, HYPHENATED for
   host-glob / device-class. The read projects a rule's selector as "<kind>:<pattern>"; the write schema
   (core/policy.ParseRuleSet) nests the SAME dimension under `match` with UNDERSCORE keys. This one table is the
   whole read→write mapping for the estate scope — built from exactly what ParseRuleSet accepts, so it can never
   silently drop or mis-bind a kind. */
const POL_SEL_KINDS = ["", "host", "host-glob", "group", "device-class", "resource"];
const POL_SEL_WRITE_KEY = { "host": "host", "host-glob": "host_glob", "group": "group", "device-class": "device_class", "resource": "resource" };

const polEdit = { step: "editing", rules: [], version: "", hasDefault: false, defaultUnknown: false, rationale: "", busy: false, outcome: null, error: null, stale: false };

/* Split a read selector string "<kind>:<pattern>" on its FIRST colon (patterns may themselves contain none),
   returning {kind, pattern}. An empty/absent selector → both empty (no estate constraint). */
function polEditSplitSelector(sel) {
  if (!sel) return { kind: "", pattern: "" };
  const s = String(sel), i = s.indexOf(":");
  if (i < 0) return { kind: s, pattern: "" };
  return { kind: s.slice(0, i), pattern: s.slice(i + 1) };
}

/* Build ONE editor card from a READ rule DTO. Every write-relevant field is carried onto the card so the
   round-trip drops nothing. min_confidence / rate_limit / reversible use `!= null` (not falsy) tests so a
   legitimately-ZERO min_confidence or a false reversible survives as a SET value rather than being read as
   "unset" — the shipped default_ruleset.json sets min_confidence:0, and dropping it would lift the floor. */
function polEditCardFromRule(r) {
  const m = (r && r.match) || {};
  const s = polEditSplitSelector(m.selector);
  return {
    id: r.id || "",
    verdict: POL_VERDICTS.indexOf(r.verdict) >= 0 ? r.verdict : "approve",
    sel_kind: s.kind || "",
    sel_pattern: s.pattern || "",
    op_class: m.op_class || "",
    argv_pattern: m.argv_pattern || "",
    territory: m.territory || "",
    reversible: m.reversible == null ? "" : (m.reversible ? "true" : "false"),
    inverse_only: m.inverse_only == null ? "" : (m.inverse_only ? "true" : "false"),
    min_confidence: r.min_confidence == null ? "" : String(r.min_confidence),
    band_mode: r.band_mode || "",
    rate_limit: r.rate_limit == null ? "" : String(r.rate_limit),
    approve_by: (r.approve_by && r.approve_by.length) ? r.approve_by.join(", ") : "",
    is_default: !!r.is_default
  };
}
function polEditBlankRule() {
  return { id: "", verdict: "approve", sel_kind: "", sel_pattern: "", op_class: "", argv_pattern: "", territory: "",
    reversible: "", inverse_only: "", min_confidence: "", band_mode: "", rate_limit: "", approve_by: "", is_default: false };
}

/* FAIL CLOSED on unknown: the editor may edit a ruleset ONLY when the read EXPLICITLY reports it has NO
   top-level default (has_default===false). A `true` blocks (a default the editor can't represent); an ABSENT
   field blocks too — a mixed-version / rolling-deploy skew where this slice-2 console briefly talks to a
   pre-slice-2 backend that never emitted has_default reads `undefined`, and treating that as editable would let
   a Save silently drop a top-level default it could not even see (turning a deny into an implicit allow). Only
   a truthful `false` unlocks editing. This is the whole point of a SAFETY guard: unknown ⇒ blocked, never open. */
function polEditDefaultBlocks(p) { return !p || p.has_default !== false; }

/* Rebuild the editor state FROM the live read (the CAS baseline). The default-representability guard decides the
   fail-closed branch and fails closed on an unknown (absent) has_default, never open. */
function polEditReset() {
  const p = liveState.policyRules || {};
  polEdit.version = p.version || "";
  polEdit.hasDefault = polEditDefaultBlocks(p);                                  // blocked unless has_default===false
  polEdit.defaultUnknown = (p.has_default !== true && p.has_default !== false);  // absent/undefined/null → "unknown"
  polEdit.rules = (p.rules || []).map(polEditCardFromRule);
  polEdit.rationale = "";
  polEdit.busy = false; polEdit.outcome = null; polEdit.error = null; polEdit.stale = false;
  polEdit.step = polEdit.hasDefault ? "blocked" : "editing";
}

function polEditIsEditable() {
  const p = liveState.policyRules;
  return !!(p && p.present && p.rules && p.rules.length);
}

function polEditRender() {
  const body = document.getElementById("polEditBody");

  body.innerHTML = "";

  else if (polEdit.step === "result") body.append(polEditResultStep());
  else if (polEdit.step === "error") body.append(polEditErrorStep());
  else body.append(polEditEditingStep());
}

/* One editable rule card. Text/select inputs write straight to the card object on input WITHOUT a re-render, so
   typing never drops caret focus (the same live idiom the mode form + tracer use). Only the STRUCTURAL controls
   (add / delete / move) mutate the list and re-render. */
function polEditCard(c, i) {
  const card = h("div", { class: "pol-edit-card" + (c.is_default ? " def" : ""), "data-idx": String(i) });
  const head = h("div", { class: "pol-edit-head" },
    h("span", { class: "pol-ord" }, c.is_default ? "def" : "#" + (i + 1)),
    h("span", { class: "pol-edit-ctrls" },
      h("button", { class: "pol-edit-mv", type: "button", "data-idx": String(i), title: "move up", "aria-label": "move rule up",
        disabled: i === 0 ? "" : null, onclick: () => polEditMove(i, -1) }, "↑"),
      h("button", { class: "pol-edit-mv", type: "button", "data-idx": String(i), title: "move down", "aria-label": "move rule down",
        disabled: i === polEdit.rules.length - 1 ? "" : null, onclick: () => polEditMove(i, 1) }, "↓"),
      h("button", { class: "pol-edit-del", type: "button", "data-idx": String(i), title: "delete rule", "aria-label": "delete rule",
        onclick: () => polEditDelete(i) }, "Delete")));
  card.append(head);

  const fld = (lbl, node) => h("label", { class: "pol-tfield" }, h("span", { class: "pol-tlbl" }, lbl), node);
  const tin = (key, ph, extra) => h("input", Object.assign(
    { class: "pol-tin", value: c[key], placeholder: ph || "", autocapitalize: "none", spellcheck: "false",
      id: "polEdit_" + key + "_" + i, "data-idx": String(i), oninput: e => { c[key] = e.target.value; } }, extra || {}));
  const pick = (key, opts, labels) => {
    const el = h("select", { class: "pol-tin", id: "polEdit_" + key + "_" + i, "data-idx": String(i),
      "aria-label": key, onchange: e => { c[key] = e.target.value; } });
    opts.forEach(o => { const op = h("option", { value: o }, (labels && labels[o]) || (o === "" ? "—" : o)); if (o === c[key]) op.selected = true; el.append(op); });
    // Never silently blank an unmappable (impossible with a valid read) selector kind — keep it visible; save refuses it.
    if (key === "sel_kind" && c.sel_kind && opts.indexOf(c.sel_kind) < 0) { const op = h("option", { value: c.sel_kind }, c.sel_kind + " (unmappable)"); op.selected = true; el.append(op); }
    return el;
  };

  const grid = h("div", { class: "pol-edit-grid" },
    fld("Rule id", tin("id", "unique id")),
    fld("Verdict", pick("verdict", POL_VERDICTS)),
    fld("Selector kind", pick("sel_kind", POL_SEL_KINDS)),
    fld("Selector pattern", tin("sel_pattern", "e.g. nl* / cisco-asa")),
    fld("Op-class", tin("op_class", "e.g. restart-service")),
    fld("Argv pattern", tin("argv_pattern", "e.g. rm -rf")),
    fld("Territory", tin("territory", "e.g. prod")),
    fld("Reversible", pick("reversible", ["", "true", "false"], { "": "— any", "true": "reversible", "false": "irreversible" })),
    fld("Inverse-only", pick("inverse_only", ["", "true", "false"], { "": "— any", "true": "inverse only", "false": "forward only" })),
    fld("Min-confidence", tin("min_confidence", "0..1", { type: "number", min: "0", max: "1", step: "0.05" })),
    fld("Band mode", pick("band_mode", POL_BAND_MODES)),
    fld("Rate limit", tin("rate_limit", "per-min", { type: "number", min: "0", step: "1" })),
    fld("Approve by", tin("approve_by", "user:a, group:b")));
  card.append(grid);

  const def = h("label", { class: "pol-edit-def" },
    h("input", { type: "checkbox", "data-idx": String(i), id: "polEdit_is_default_" + i, checked: c.is_default ? "" : null,
      onchange: e => { c.is_default = e.target.checked; } }),
    h("span", {}, "global-default rule (contributes params; needs no match)"));
  card.append(def);
  return card;
}

  const t = polEdit.rules[i]; polEdit.rules[i] = polEdit.rules[j]; polEdit.rules[j] = t;

/* Build ONE write-schema rule doc from an editor card — the read→write mapping that must drop nothing:
   the hyphenated selector kind becomes its UNDERSCORE match key, params NEST under `params`, and every SET
   field round-trips (empty/unset fields are omitted so the document matches the canonical shape). Returns
   {rule} or {error} — an unmappable selector kind FAILS CLOSED (refuses), it never silently drops the scope. */
function polEditRuleDoc(c) {
  const id = (c.id || "").trim();
  if (!id) return { error: "every rule needs an id" };
  if (POL_VERDICTS.indexOf(c.verdict) < 0) return { error: "rule " + id + ": invalid verdict" };
  const rule = { id: id, verdict: c.verdict };
  const match = {};
  if (c.sel_kind) {
    const key = POL_SEL_WRITE_KEY[c.sel_kind];
    if (!key) return { error: "rule " + id + ": selector kind '" + c.sel_kind + "' cannot be represented — refusing to drop it" };
    const pat = (c.sel_pattern || "").trim();
    if (pat) match[key] = pat;
  }
  if ((c.op_class || "").trim()) match.op_class = c.op_class.trim();
  if ((c.argv_pattern || "").trim()) match.argv_pattern = c.argv_pattern.trim();
  if ((c.territory || "").trim()) match.territory = c.territory.trim();
  if (c.reversible === "true") match.reversible = true;
  else if (c.reversible === "false") match.reversible = false;
  if (c.inverse_only === "true") match.inverse_only = true;
  else if (c.inverse_only === "false") match.inverse_only = false;
  if (Object.keys(match).length) rule.match = match;
  const params = {};
  if (c.min_confidence !== "" && c.min_confidence != null) {
    const n = Number(c.min_confidence);
    if (isNaN(n)) return { error: "rule " + id + ": min_confidence must be a number" };
    params.min_confidence = n;
  }
  if (c.band_mode) params.band_mode = c.band_mode;
  if (c.rate_limit !== "" && c.rate_limit != null) {
    const n = parseInt(c.rate_limit, 10);
    if (isNaN(n)) return { error: "rule " + id + ": rate_limit must be an integer" };
    params.rate_limit = n;
  }
  if (Object.keys(params).length) rule.params = params;
  const approvers = (c.approve_by || "").split(",").map(s => s.trim()).filter(Boolean);
  if (approvers.length) rule.approve_by = approvers;
  if (c.is_default) rule.is_default = true;
  return { rule: rule };
}
/* Compose the FULL replacement document from every card, in order. NO top-level `default` is ever emitted —
   the editor refuses to open a ruleset that has one (hasDefault), so a round-trip can never carry a phantom or
   drop a real default. */
function polEditBuildDocument() {
  const rules = [];
  for (const c of polEdit.rules) {
    const r = polEditRuleDoc(c);
    if (r.error) return { error: r.error };
    rules.push(r.rule);
  }
  return { document: { rules: rules } };
}

/* Map a ruleset-write failure to an honest sentence + flags — grounded in the REAL contract (core/httpapi
   ruleset_write.go + adminWriteGuard + AuthAdminSession): 401 = absent/expired/plain session → drive the shared
   step-up; 403 = cross-origin OR not admin-authorized → an honest terminal refusal (elevation cannot fix it);
   409 = stale expected_version ("changed underneath you"); 400 = the validator refused — its VERBATIM message
   is shown, never a success claim; 503/404/405 = the worker did not accept the write (fail-closed). */
async function polEditPostFail(r) {
  let server = ""; try { server = (await r.text()).trim().slice(0, 400); } catch (e) {}
  if (r.status === 401) return { elevate: true, msg: "admin elevation required — step up to write the ruleset" };
  if (r.status === 403) return { msg: "refused: " + (server || "this session may not write the ruleset (not admin-authorized, or a cross-origin write)") };
  if (r.status === 409) return { stale: true, msg: "the ruleset changed underneath you — " + (server || "expected_version no longer matches the active ruleset") + ". Re-read the current ruleset and reapply your edits; nothing was overwritten." };
  if (r.status === 400) return { msg: server || "refused: malformed ruleset" }; // VERBATIM backend validator text
  if (r.status === 429) return { msg: "rate limited — wait a moment and retry" };
  if (r.status === 503 || r.status === 404 || r.status === 405) return { msg: "the ruleset-write path is unavailable — the worker did not accept the write (fail-closed); the ruleset was NOT changed" };
  return { msg: "ruleset write failed (" + r.status + ")" + (server ? " — " + server : "") };
}

/* Re-read the live ruleset after a committed write (so the table + the CAS baseline tell the truth). On a
   failed re-read the prior value is KEPT — the result step already carries the worker's committed outcome. */
async function polEditRefreshRules() { try { liveState.policyRules = await liveGet("/v1/policy/rules"); } catch (e) {} }

async function polEditSubmit() {
  // Defense in depth, FAIL CLOSED on unknown: never emit a document unless the live read EXPLICITLY reports the
  // ruleset has no top-level default (has_default===false). A `true` OR an absent field (a pre-slice-2 backend
  // that never emitted it) both block — the editor must never POST a replacement that could silently drop a
  // default. Re-derived from the LIVE read here, not just the reset-time flag, so the guard bites at the POST.
  if (polEdit.hasDefault || polEditDefaultBlocks(liveState.policyRules)) { polEdit.error = "This ruleset has — or the control plane didn't confirm it lacks — a top-level default the editor can't represent; refusing to save (use the ruleset API)."; polEdit.stale = false; polEdit.step = "error"; polEditRender(); return; }
  if (!polEdit.rationale.trim() || !polEdit.rules.length) return; // gated by the form; belt-and-braces
  const built = polEditBuildDocument();
  if (built.error) { polEdit.error = "refused (client-side): " + built.error; polEdit.stale = false; polEdit.step = "error"; polEditRender(); return; }

  polEdit.step = "saving"; polEdit.busy = true; polEdit.error = null; polEditRender();
  const payload = { document: built.document, expected_version: polEdit.version, rationale: polEdit.rationale.trim() };
  let r = null;
  try {
    r = await fetch("/api/v1/policy/ruleset", { method: "POST", credentials: "same-origin",
      headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) });
  } catch (e) {}
  polEdit.busy = false;

  if (r && r.ok) {
    let out = {}; try { out = await r.json(); } catch (e) {}
    polEdit.outcome = out; polEdit.error = null; polEdit.step = "result"; polEditRender();
    await polEditRefreshRules(); // re-read so the table + CAS baseline update honestly on close
    return;
  }
  const f = r ? await polEditPostFail(r) : { msg: "network error — the control plane did not answer; the ruleset was NOT changed. Retry." };
  if (f.elevate) {
    // Reuse the SHARED admin step-up (config/secret + mode-change use the same modal + /api/v1/session/elevate).
    // Close this dialog first so only one modal is live; on a successful step-up re-open PRESERVING the edit —
    // nothing auto-submits, so the operator's Save is a deliberate second click.
    polEditClose();
    if (typeof cfgElevateOverlay === "function") cfgElevateOverlay("Writing the ruleset needs an elevated session.", () => polEditReopen());
    return;
  }
  polEdit.error = f.msg; polEdit.stale = !!f.stale; polEdit.step = "error"; polEditRender();
}

function polEditResultStep() {
  const o = polEdit.outcome || {};
  const wrap = h("div", { style: "padding:10px 20px 20px;display:flex;flex-direction:column;gap:12px" });
  wrap.append(h("div", { class: "pol-mconfirm" },

const POL_PROMOTE_DEFAULT = 5; // engine DefaultPromoteThreshold — labeled as the default, never asserted per-class.
function polGradSection() {
  const strip = h("div", { class: "pol-strip" });
  strip.append(polSec("LIVE · graduation ladder", "Earned autonomy — clean runs toward promotion"));
  const g = liveState.policyGraduation;
  if (g === undefined) { strip.append(polErr("Connecting to the policy engine…")); return strip; }
  if (g === null) { strip.append(polErr("/v1/policy/graduation is not served by this control plane yet (fail-closed).")); return strip; }
  if (!g.length) {
    strip.append(polNone("No op-class has graduated yet", "The ladder is empty — an op-class appears here once the engine records its first run outcome. Autonomy is earned, never assumed."));
    return strip;
  }

  const t = h("table", { class: "tbl" });
  t.append(h("thead", {}, h("tr", {}, ...["Op-class", "Level", "Progress", "Domain", "Last outcome", "Updated"].map(x => h("th", {}, x)))));
  const tb = h("tbody");
  g.forEach(c => {
    const lvl = String(c.level || "").toLowerCase();
    const auto = lvl === "auto";
    const notice = lvl === "auto_notice";
    /* THE HONEST CEILING (spec/028 REQ-2808, ADR-0016 decision 2). An overlay class that finished the
       second climb is HELD at auto_notice with its streak PINNED at the threshold — not reset, precisely
       so this state is distinguishable from "still climbing". The server derives ceiling_held from
       (level, streak, embedded); rendering a fraction here would promise a promotion that cannot arrive
       without an embed-export MR, so the held state is captioned in words instead. */
    const held = !!c.ceiling_held;
    /* Which climb is this class on, and against WHICH bar? The two streaks are counted separately and
       must never be rendered against each other's threshold — 3 runs into the second climb is not
       "3/5 of the way to acting at all". notice_threshold is served per class; clean_run_count's bar is
       the engine default (a class's own may differ, which the caption says rather than guesses). */
    const n = notice ? (c.notice_run_count || 0) : (c.clean_run_count || 0);
    const bar = notice ? (c.notice_threshold || POL_PROMOTE_DEFAULT) : POL_PROMOTE_DEFAULT;
    const frac = auto ? 1 : Math.min(n, bar) / (bar || 1);
    const fill = auto ? "var(--map)" : (notice ? "var(--warn)" : "var(--match)");
    let prog;
    if (held) {
      prog = h("span", { class: "pol-held", title: "the second climb is complete; the silent rung is reserved for the embedded, lockstep-hashed registry" },
        "earned auto — awaiting an embed-export MR");
    } else if (auto) {

    } else {
      prog = h("span", { class: "pol-grad", title: n + " consecutive verified-clean runs toward " + (notice ? "auto (threshold " + bar + ")" : "auto-notice (default threshold " + bar + ")") },
        h("span", { class: "pol-grad-bar" }, h("span", { class: "pol-grad-fill", style: "width:" + (frac * 100).toFixed(0) + "%;background:" + fill })),
        h("span", { class: "pol-grad-n" }, n + " / " + bar));
    }
    /* Which tamper domain grants this class is the ONLY thing that decides whether the silent rung is
       reachable at all, so it is a column rather than a footnote. */
    const domain = c.embedded
      ? h("span", { class: "pol-pill", style: "color:var(--map);border-color:color-mix(in srgb,var(--map) 45%,transparent)", title: "go:embed, lockstep-hashed — a code release admitted it" }, "EMBEDDED")
      : h("span", { class: "pol-pill", style: "color:var(--ink3);border-color:var(--line2)", title: "runtime overlay — ratified by an operator; capped at auto-notice" }, "OVERLAY");
    tb.append(h("tr", {},
      h("td", { class: "m", style: "color:var(--map)" }, c.op_class || "—"),
      h("td", {}, polLevel(c.level)),
      h("td", {}, prog),
      h("td", {}, domain),
      h("td", {}, polOutcome(c.last_outcome)),
      h("td", { class: "m", style: "color:var(--ink3)" }, polRel(c.updated_at))));
  });
  t.append(tb);
  strip.append(polCard(t));
  strip.append(h("div", { class: "pol-cap" }, "Three rungs: approve asks first, auto-notice acts and tells you, auto acts silently. Progress counts consecutive verified-clean runs toward the NEXT rung against that climb's own bar. A deviation demotes to approve and resets the streak. An OVERLAY class can climb to auto-notice and no further — the silent rung is reserved for the embedded, lockstep-hashed registry, and export-embed generates the MR that would carry it there."));
  return strip;
}

/* ---- section 5: packet-tracer — "may TG act on host X with op-class Y?" (TG-105 slice 2) -------
   LIVE. POSTs a HYPOTHETICAL candidate to /api/v1/policy/trace and renders the composed verdict the
   worker's REAL Policy Engine returns over Temporal — the same composer the interceptor consults, run
   BARE (no audit row, no rate governor). READ-ONLY (AuthReadOnly): it asks "may I?", never the act — it
   actuates nothing and writes no policy_decision row. The states are real and only real: idle (nothing
   traced yet) · loading · a live result · an honest error (401 re-gates to the auth overlay, 400 shows
   the server's validation message, 503/network say the engine was unreachable) — NEVER a fabricated
   verdict (INV-15). The response is non-secret authZ metadata only (verdicts, a rule id, a band, a
   bundle fingerprint); no argv or host credential can appear in it (INV-13). */
const POL_TRACE_MODES = ["Shadow", "HITL", "Semi-auto", "Full-auto"];
const polTrace = { op_class: "", host: "", resource: "", confidence: "0.7", band: "AUTO", mode: null,
                   loading: false, result: undefined, error: null };

/* The LIVE autonomy mode the console already fetched (liveState.policyMode), normalized to the tracer's
   option set — or null when unknown / fail-closed. CRITICAL UX (review finding): an operator who leaves
   the mode untouched is asking "what would happen RIGHT NOW", so the form defaults to this, NOT to the
   fail-closed Shadow; only a genuinely-unknown live mode falls back to Shadow, and the caption says
   which of the two it is rather than pretending Shadow is the live posture. */
function polTraceLiveMode() {
  const m = liveState.policyMode;
  if (!m || !m.mode) return null;
  return POL_TRACE_MODES.find(x => x.toLowerCase() === String(m.mode).toLowerCase()) || null;
}
/* The mode that WILL be sent: the operator's explicit pick, else the live mode, else Shadow. */
function polTraceEffectiveMode() { return polTrace.mode || polTraceLiveMode() || "Shadow"; }

/* Map an HTTP failure to an honest sentence + whether it is a session-expiry that must re-gate. Mirrors
   the skills / manifest / candidates write-lanes: a 401/403 re-prompts the login overlay (the operator
   session lapsed mid-trace), a 400 surfaces the server's own validation message, a 503/404/405 is the
   engine unreachable (fail-closed, no verdict), and anything else names the status. NEVER a verdict. */
async function polTraceFail(r) {
  let server = ""; try { server = (await r.text()).trim().slice(0, 300); } catch (e) {}
  if (r.status === 401 || r.status === 403) return { login: true, msg: "operator session required — sign in to trace a policy decision" };
  if (r.status === 400) return { msg: "refused: " + (server || "describe the candidate action — at least an op-class or a host is required") };
  if (r.status === 503 || r.status === 404 || r.status === 405) return { msg: "trace unavailable — the worker Decide engine is not reachable (fail-closed); no verdict is shown" };
  return { msg: "trace failed (" + r.status + ")" + (server ? " — " + server : "") };
}

async function polRunTrace() {
  // The tracer is named for "op-class Y on host X" — require both before asking. (The server accepts
  // either alone, but a half-described candidate here is a mistake, not a question, so gate it client-side.)
  if (!polTrace.op_class.trim() || !polTrace.host.trim()) {
    polTrace.error = "describe the candidate — op-class and host are both required";
    polTrace.result = undefined; polRerenderView(); return;
  }
  polTrace.loading = true; polTrace.error = null; polTrace.result = undefined; polRerenderView();

  const body = { op_class: polTrace.op_class.trim(), host: polTrace.host.trim(),
    band: polTrace.band, mode: polTraceEffectiveMode() };
  if (polTrace.resource.trim()) body.resource = polTrace.resource.trim();
  const conf = parseFloat(polTrace.confidence);
  if (!isNaN(conf)) body.confidence = conf;

  let r = null;
  try {
    r = await fetch("/api/v1/policy/trace", { method: "POST", credentials: "same-origin",
      headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
  } catch (e) {}
  polTrace.loading = false;

  if (r && r.ok) {
    let out = {}; try { out = await r.json(); } catch (e) {}
    polTrace.result = out; polTrace.error = null; polRerenderView(); return;
  }
  const f = r ? await polTraceFail(r) : { msg: "network error — the control plane did not answer; retry" };
  polTrace.error = f.msg; polTrace.result = undefined; polRerenderView();
  // Session lapsed mid-trace → re-show the full-screen auth gate (the same call the skills write-lane makes).
  if (f.login) liveLoginOverlay();
}

/* The composed verdict as a CHIP — ASA packet-tracer allow/deny semantics answering "may I act?": auto =
   permitted to act (chip-ok), approve = permitted but human-gated (chip-warn), deny = blocked (chip-bad).
   This is the SAME chip-ok/chip-warn/chip-bad polarity the mode hero already uses (NO AUTO-ACTUATION = ok,
   MAY AUTO-ACTUATE = bad). An unrecognised verdict is a neutral pill, never a green pass. */
function polTraceVerdictChip(v) {
  const t = String(v || "").toLowerCase();
  if (t === "auto") return h("span", { class: "chip-ok" }, "AUTO");
  if (t === "approve") return h("span", { class: "chip-warn" }, "APPROVE");
  if (t === "deny") return h("span", { class: "chip-bad" }, "DENY");
  return h("span", { class: "pol-pill", style: "color:var(--ink3);border-color:var(--line2)" }, (v || "—").toUpperCase());
}

        : h("span", { style: "color:var(--ink4)" }, "—"))));

  // approve_by — only meaningful when the verdict resolved to approve (who must vote).
  if (String(res.verdict).toLowerCase() === "approve" && res.approve_by && res.approve_by.length) {
    box.append(h("div", { class: "pol-tmeta" },
      h("div", {}, h("span", { class: "pol-tk" }, "approve by"),
        h("span", { class: "pol-approve" }, res.approve_by.join(", ")))));
  }

  // matched_rules — the full deny-overrides provenance, rendered with the SAME polVerdict pill the rules
  // table and decision log use, in its own overflow-x container so the page body never scrolls sideways.
  if (res.matched_rules && res.matched_rules.length) {
    const t = h("table", { class: "tbl" });
    t.append(h("thead", {}, h("tr", {}, ...["Rule", "Verdict"].map(x => h("th", {}, x)))));
    const tb = h("tbody");
    res.matched_rules.forEach(m => tb.append(h("tr", {},
      h("td", { class: "m" }, m.rule_id ? h("span", { style: "color:var(--territory)" }, m.rule_id) : h("span", { style: "color:var(--ink4)" }, "—")),
      h("td", {}, polVerdict(m.verdict)))));
    t.append(tb);
    box.append(h("div", {},
      h("div", { class: "pol-tk", style: "margin-bottom:6px" }, "matched rules · deny-overrides"),
      h("div", { class: "pol-scroll" }, t)));
  }

  // The reason — the multi-clause base→confidence/rate→band→graduation→floor explanation, read as prose.
  if (res.reason) {
    box.append(h("div", { class: "pol-treason" },
      h("span", { class: "pol-tk" }, "why"),
      h("p", { class: "pol-posture", style: "margin:4px 0 0" }, res.reason)));
  }

  // MANDATORY honesty caveat — rate_governor_simulated is always false: the trace runs the bare composer,
  // so it does NOT model the runtime rate governor. A composed `auto` can still be rate-clamped to
  // `approve` at real actuation. Surfaced verbatim from the server's flag, never hidden.
  box.append(h("div", { class: "pol-tcaveat" },
    h("b", {}, "Rate governor not simulated"),
    " (rate_governor_simulated=" + String(!!res.rate_governor_simulated) + "). This trace runs the bare composed engine — it does not model the runtime rate governor, so a composed ",
    h("b", { style: "color:var(--map)" }, "auto"), " may still be rate-clamped to ",
    h("b", { style: "color:var(--partial)" }, "approve"), " at real actuation time."));

  return box;
}

function polTraceSection() {
  const strip = h("div", { class: "pol-strip" });
  strip.append(polSec("LIVE · packet-tracer", "“May TG act on host X with op-class Y?” — trace a candidate through mode → rules → graduation → band"));
  strip.append(polTraceForm());

  if (polTrace.loading) {
    strip.append(h("div", { class: "empty", style: "padding:22px" },
      h("div", { class: "desc" }, "Tracing the candidate through the live policy engine…")));
    return strip;
  }
  if (polTrace.error) {
    strip.append(h("div", { class: "wrapcard pol-terr" },
      h("span", { class: "chip-bad" }, "TRACE FAILED"),
      h("p", { class: "pol-note", style: "margin:8px 0 0" }, polTrace.error)));
    return strip;
  }
  if (polTrace.result === undefined) {
    strip.append(polNone("No trace run yet",
      "Describe a candidate action above and run Trace — the worker's real Policy Engine composes the verdict (auto / approve / deny), which rule matched, the effective band and why. Nothing is actuated and no decision row is written."));
    return strip;
  }
  strip.append(polTraceResult(polTrace.result));
  return strip;
}
