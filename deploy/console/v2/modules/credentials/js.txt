/* ============================================================================
   CREDENTIALS module (cred-) — the credential engine's operator surface (TG-107).
   Answers "where do credentials come from?", "what can TG reach?", and "who
   resolved, via which source + identity?" — all from the REAL persisted
   projections served by the read-only credentials API. authN metadata only:
   schemes, login users, counts, provenance. No secret material is ever fetched
   or shown (INV-13); the API has no field that could carry one.
   TG-109 adds the WRITE half: per-source Sync now (worker lane, outcome shown
   verbatim incl. the starved sentence), Configure (the EXISTING generated
   module dialog — the inline/native sources honestly have none), the source's
   published precedence rank, and the native-db mapping rules (elevated
   trace-read list, ledger-first add/delete with the validator's text verbatim).
   Epistemics/theming: the machine plane is MAP-indigo, the human plane is
   TERRITORY-steel; outcomes reuse the console's ok/partial/deviation semantics.
   ============================================================================ */

/* relative time from an RFC3339 string; honest "never"/passthrough on empty/unparseable. */
function credRel(iso) {
  if (!iso) return "never";
  const t = Date.parse(iso);
  if (isNaN(t)) return String(iso);
  let s = Math.floor((Date.now() - t) / 1000);
  if (s < 0) return "just now";
  if (s < 60) return s + "s ago";
  const m = Math.floor(s / 60); if (m < 60) return m + "m ago";
  const hh = Math.floor(m / 60); if (hh < 24) return hh + "h ago";
  const d = Math.floor(hh / 24); if (d < 30) return d + "d ago";
  const mo = Math.floor(d / 30); if (mo < 12) return mo + "mo ago";
  return Math.floor(mo / 12) + "y ago";
}

/* sync-source outcome chip — ok / partial / failed, reusing the console chip palette. */
function credSrcChip(o) {
  const v = String(o || "").toLowerCase();
  const cls = v === "ok" ? "chip-ok" : v === "partial" ? "chip-warn" : "chip-bad";
  return h("span", { class: cls }, (o || "unknown").toUpperCase());
}

/* resolution outcome chip — resolved / ambiguous / (unresolved|refused). err text becomes a tooltip. */
function credResChip(o, err) {
  const v = String(o || "").toLowerCase();
  const cls = v === "resolved" ? "chip-ok" : v === "ambiguous" ? "chip-warn" : "chip-bad";
  return h("span", { class: cls, title: err || "" }, (o || "—").toUpperCase());
}

/* plane pill — machine (map-indigo) vs human (territory-steel), the map-vs-territory language. */
function credPlane(pl) {
  const v = String(pl || "").toLowerCase();
  const tint = v === "machine" ? "var(--map)" : v === "human" ? "var(--territory)" : "var(--ink3)";
  return h("span", { class: "cred-plane", style: "color:" + tint + ";border-color:color-mix(in srgb," + tint + " 45%,transparent)" }, v || "—");
}

function credErr(msg) { return h("div", { class: "empty", style: "padding:22px" }, h("div", { class: "desc" }, msg)); }
function credNone(big, desc) { return h("div", { class: "empty", style: "padding:34px 20px" }, h("div", { class: "big" }, big), h("div", { class: "desc" }, desc)); }

function credSec(lbl, title, extra) {
  return h("div", { class: "sec" },
    h("span", { class: "lbl", style: "color:var(--match)" }, lbl),
    h("h3", {}, title),
    h("span", { class: "rule" }),
    extra || null);
}

/* a table wrapped in its own horizontal-scroll container — the page body never scrolls sideways. */
function credCard(table) { return h("div", { class: "wrapcard" }, h("div", { class: "cred-scroll" }, table)); }

/* ---- TG-109 write half: shared refusal→sentence mapper ----------------------------------------
   Mirrors polEditPostFail: wherever the backend's text is load-bearing (a validator refusal, a typed
   404) it is shown VERBATIM, never paraphrased into a success or a vaguer failure. */
async function credWriteFail(r) {
  let server = ""; try { server = (await r.text()).trim().slice(0, 400); } catch (e) {}
  if (r.status === 401) return { elevate: true, msg: "admin elevation required — step up, then press the button again (nothing auto-submits)" };
  if (r.status === 403) return { msg: "refused: " + (server || "this session may not write here (not admin-authorized, or a cross-origin write)") };
  if (r.status === 404) return { gone: true, msg: server || "not found" };
  if (r.status === 429) return { msg: "rate limited — wait a moment and retry" };
  if (r.status === 400) return { msg: server || "refused" }; // VERBATIM backend validator text
  if (r.status === 503 || r.status === 405) return { msg: "this write path is not deployed on this control plane (fail-closed) — nothing was changed" };
  return { msg: "write failed (" + r.status + ")" + (server ? " — " + server : "") };
}

/* After a successful admin step-up driven from this surface: re-read what the elevation unlocks.
   ALWAYS passed to cfgElevateOverlay — its default callback routes to #secrets, which would bounce the
   operator off this view. */
function credAfterElevate() {
  if (credNat.phase === "err") { credNativeLoad(true); return; }
  credRerenderView();
}

/* ---- per-source "Sync now" (POST /v1/credentials/sources/{id}/sync) ---------------------------
   Outcome state lives on the module (source_id → busy/outcome/note) so the 60s live-refresh re-render
   rebuilds the same truth instead of wiping it. One in-flight click per source ⇒ one POST. */
const credSyncState = {};

/* one source's sync outcome, VERBATIM: the worker's summary + detail sentences ARE the operator's
   answer. The starved sentence is load-bearing — a starved source (synced OK, ZERO bindings) reports
   0 drift and looks converged everywhere else; only this wording separates it from success. */
function credSyncOutcome(out) {
  const chip = out.ok
    ? (out.starved ? h("span", { class: "chip-warn" }, "SYNCED · STARVED") : h("span", { class: "chip-ok" }, "SYNCED"))
    : h("span", { class: "chip-bad" }, "SYNC FAILED");
  const wrap = h("div", { class: "cred-syncout" },
    h("div", { class: "cred-syncout-head" },
      chip,
      h("span", { style: "color:var(--ink)" }, out.summary || ""),
      h("span", { class: "m", style: "color:var(--ink4)" },
        "+" + (out.added || 0) + " ~" + (out.changed || 0) + " -" + (out.removed || 0)
        + " · " + (out.entries || 0) + " entries"
        + (out.elapsed_ms != null ? " · " + out.elapsed_ms + "ms" : ""))));
  if (out.detail) wrap.append(h("div", { class: "desc", style: "margin-top:4px;font-size:11px" }, out.detail));
  return wrap;
}

async function credSyncNow(r) {
  const id = r.source_id || "";
  if (credSyncState[id] && credSyncState[id].kind === "busy") return; // one click in flight ⇒ one POST
  credSyncState[id] = { kind: "busy" };
  credRerenderView();
  let resp = null;
  try {
    resp = await fetch("/api/v1/credentials/sources/" + encodeURIComponent(id) + "/sync",
      { method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json" }, body: "{}" });
  } catch (e) {}
  if (resp && resp.ok) {
    const out = await resp.json().catch(() => null);
    credSyncState[id] = out ? { kind: "outcome", out: out }
      : { kind: "note", tone: "bad", msg: "the control plane answered, but the outcome did not parse" };
  } else if (resp && resp.status === 502) {
    // Same honesty as the module TEST button: could-not-run is a TG-side fault, NOT a verdict on the source.
    credSyncState[id] = { kind: "note", tone: "bad",
      msg: "the sync could not be run — the worker did not answer. This is a TG-side problem, not a verdict on the source." };
  } else {
    const f = resp ? await credWriteFail(resp) : { msg: "network error — the control plane did not answer; retry" };
    credSyncState[id] = { kind: "note", tone: f.elevate ? "dim" : "bad", msg: f.msg };
    if (f.elevate && typeof cfgElevateOverlay === "function")
      // the polEditSubmit discipline: only ONE modal live, and nothing auto-submits after the
      // step-up — Sync now is a deliberate second click.
      cfgElevateOverlay("Running a source sync needs an elevated session.", credAfterElevate);
  }
  credRerenderView();
}

/* ---- per-source "Configure": the EXISTING generated module dialog (modCfgCard), reused verbatim.
   Only the connectors that publish a schema have a dialog; the inline/native sources (native-hostdiag,
   native-db) deliberately publish none and that absence is rendered honestly, never papered over. */
function credCfgClose() {
  const el = document.getElementById("credCfgDlg");
  if (!el) return;
  if (typeof overlayClose === "function") overlayClose(el);
  el.remove();
}
function credConfigure(sourceId) {
  if (document.getElementById("credCfgDlg")) return;
  const body = h("div", { id: "credCfgBody", style: "padding:4px 20px 20px;overflow:auto" });
  const scrim = h("div", { class: "pal-scrim open", id: "credCfgDlg", style: "align-items:flex-start;padding-top:4vh" },
    h("div", { class: "pal", role: "dialog", "aria-label": "Configure credential source",
      style: "width:calc(100% - 32px);max-width:760px;max-height:calc(100vh - 8vh);display:flex;flex-direction:column" },
      h("div", { style: "padding:18px 20px 4px;display:flex;align-items:baseline;gap:10px" },
        h("div", {},
          h("div", { class: "lbl", style: "color:var(--map)" }, "Credential source · configuration"),
          h("h3", { style: "margin-top:6px;font-size:16px" }, sourceId)),
        h("button", { class: "btn", type: "button", style: "margin-left:auto", onclick: () => credCfgClose() }, "Close")),
      body));
  const render = () => {
    body.innerHTML = "";
    if (typeof modCfgState === "undefined" || !modCfgState.schema) {
      body.append(h("div", { class: "empty", style: "padding:22px" }, h("div", { class: "desc" },
        (typeof modCfgState !== "undefined" && modCfgState.tried)
          ? "/v1/modules/schema is unavailable — the generated dialog cannot be built (nothing is invented in its place)."
          : "reading the module schema…")));
      return;
    }
    const mod = (modCfgState.schema.modules || []).find(m => m && m.surface === "credsource" && m.source_type === sourceId);
    if (!mod) {
      body.append(h("div", { class: "empty", style: "padding:26px 18px" },
        h("div", { class: "big" }, "no dialog (inline/native source)"),
        h("div", { class: "desc" }, "This source is built into TG — its rules are operator-authored here (the native mapping "
          + "section) or in config, not pulled from a credentialed connector, so it publishes no configuration schema and "
          + "there is no dialog to open.")));
      return;
    }
    const cfgValues = {};
    (Array.isArray(liveState.config) ? liveState.config : []).forEach(k => { if (k && k.name) cfgValues[k.name] = k.value; });
    if (typeof modCfgCard === "function") body.append(modCfgCard(mod, cfgValues)); // the EXISTING generated dialog
    else body.append(h("div", { class: "empty", style: "padding:22px" }, h("div", { class: "desc" }, "the module dialog renderer is not loaded in this bundle")));
  };
  document.body.append(scrim);
  render();
  if (typeof modCfgEnsureSchema === "function") modCfgEnsureSchema(render);
  if (typeof overlayOpen === "function") overlayOpen(scrim, { label: "Configure credential source",
    onClose: () => { const x = document.getElementById("credCfgDlg"); if (x) x.remove(); } });
}

/* ---- section 1: sync sources — config + last-synced/drift + per-source actions ---------------- */
function credSourcesSection() {
  const strip = h("div", { class: "cred-strip" });
  strip.append(credSec("LIVE · sync sources", "Where credentials come from — last sync & drift"));
  const s = liveState.credSources;
  if (s === undefined) { strip.append(credErr("Connecting to the credential engine…")); return strip; }
  if (s === null) { strip.append(credErr("/v1/credentials/sources is not served by this control plane yet (fail-closed).")); return strip; }
  if (!s.length) { strip.append(credNone("No credential sources configured", "Nothing has been imported or registered yet — add a sync source to give TG reach.")); return strip; }

  const t = h("table", { class: "tbl" });
  t.append(h("thead", {}, h("tr", {}, ...["Source", "Plane", "Rank", "Outcome", "Last synced", "Drift (added/changed/removed)", "Covered", "Detail", "Actions"].map(x => h("th", {}, x)))));
  const tb = h("tbody");
  s.forEach(r => {
    const id = r.source_id || "";
    const failed = String(r.outcome || "").toLowerCase() !== "ok";
    const drift = r.drifted
      ? h("span", {}, h("span", { style: "color:var(--match)" }, "+" + (r.added || 0)), " ",
          h("span", { style: "color:var(--partial)" }, "~" + (r.changed || 0)), " ",
          h("span", { style: "color:var(--deviation)" }, "-" + (r.removed || 0)))
      : h("span", { style: "color:var(--ink4)" }, "clean");
    // Precedence is the source's published compiled resolution rank. The field is omitempty and 0 is
    // NOT a valid rank (compiled ranks start at 10), so an absent field — a pre-0087 projection that
    // never published it — renders the honest "unpublished", never 0-as-rank.
    const rank = (r.precedence == null || r.precedence === 0)
      ? h("span", { style: "color:var(--ink4)", title: "this control plane's projection predates precedence publication — the worker holds the rank but does not publish it here" }, "unpublished")
      : h("span", { class: "m", style: "color:var(--map)", title: "compiled resolution rank — lower wins" }, "#" + r.precedence);
    const st = credSyncState[id];
    const syncBtn = h("button", { class: "btn cred-actbtn", type: "button", "aria-label": "Sync " + (id || "source") + " now" }, "SYNC NOW");
    if (st && st.kind === "busy") { syncBtn.disabled = true; syncBtn.setAttribute("aria-disabled", "true"); }
    syncBtn.addEventListener("click", () => credSyncNow(r));
    const cfgBtn = h("button", { class: "btn cred-actbtn", type: "button", "aria-label": "Configure " + (id || "source") }, "CONFIGURE");
    cfgBtn.addEventListener("click", () => credConfigure(id));
    tb.append(h("tr", { class: failed ? "cred-row-fail" : "" },
      h("td", { class: "m terr" }, id || "—"),
      h("td", {}, credPlane(r.plane)),
      h("td", {}, rank),
      h("td", {}, credSrcChip(r.outcome)),
      h("td", { class: "m", style: "color:var(--ink3)" }, r.last_synced_at ? credRel(r.last_synced_at) : "never"),
      h("td", { class: "m" }, drift),
      h("td", { class: "m", style: "color:var(--territory)" }, String(r.covered_targets != null ? r.covered_targets : 0)),
      h("td", { class: "m", style: failed ? "color:var(--deviation)" : "color:var(--ink4)" }, r.err ? r.err : "—"),
      h("td", { class: "cred-actcell" }, syncBtn, cfgBtn)));
    // The last sync-now outcome (or its refusal) lands in a full-width row directly under the source,
    // rebuilt from module state on every re-render so the live-refresh cycle cannot wipe it.
    if (st && st.kind !== undefined) {
      const cell = h("td", { colspan: "9", class: "cred-syncrow-cell" });
      if (st.kind === "busy") cell.append(h("span", { class: "desc", style: "font-size:11px" }, "running the real sync…"));
      else if (st.kind === "outcome") cell.append(credSyncOutcome(st.out));
      else cell.append(h("span", { class: "desc", style: "font-size:11px;color:" + (st.tone === "bad" ? "var(--deviation)" : "var(--ink3)") }, st.msg));
      tb.append(h("tr", { class: "cred-syncrow" }, cell));
    }
  });
  t.append(tb);
  strip.append(credCard(t));
  return strip;
}

/* ---- section 1b: the NATIVE mapping rules (TG-109 write half) ---------------------------------
   GET /v1/credentials/native sits at the elevated trace-read tier — each row pairs a target selector
   with the SecretRef REFERENCE that unlocks it — while this module loads for every read-only session.
   The 403 is therefore an EXPECTED state here and renders honestly with the step-up affordance, never
   as a generic error. SecretRef fields render as the reference STRING only (non-secret by INV-13);
   there is deliberately NO control anywhere on this surface that fetches or displays a resolved value.
   Writes go through the worker's ledger-first lane; the validator's refusal text is shown verbatim. */
let credNat = { phase: "unloaded", rows: null, total: 0, status: 0, timeout: false };
let credNatForm = { entry: "", rationale: "", busy: false };
let credNatMsg = { text: "", tone: "dim" };   // the one write-outcome line (add + delete share it)
let credNatDel = { id: 0, rationale: "", note: "" }; // one open delete prompt at a time

function credNativeLoad(force) {
  if (credNat.phase === "loading") return;
  if (!force && credNat.phase !== "unloaded") return;
  credNat = { phase: "loading", rows: null, total: 0, status: 0, timeout: false };
  (async () => {
    try {
      const p = await liveGet("/v1/credentials/native");
      credNat = { phase: "ok", rows: p.rules || [], total: p.total != null ? p.total : (p.rules || []).length, status: 200, timeout: false };
    } catch (e) {
      credNat = { phase: "err", rows: null, total: 0, status: (e && e.status) || 0, timeout: !!(e && e.timeout) };
    }
    credRerenderView();
  })();
}

async function credNativeAdd() {
  if (credNatForm.busy) return; // one in-flight add ⇒ one POST
  const entry = credNatForm.entry.trim(), rationale = credNatForm.rationale.trim();
  if (!entry || !rationale) {
    credNatMsg = { text: "entry and rationale are both required — every native rule states why it exists", tone: "bad" };
    credRerenderView();
    return;
  }
  credNatForm.busy = true;
  credNatMsg = { text: "recording…", tone: "dim" };
  credRerenderView();
  let resp = null;
  try {
    resp = await fetch("/api/v1/credentials/native/rules", { method: "POST", credentials: "same-origin",
      headers: { "Content-Type": "application/json" }, body: JSON.stringify({ entry: entry, rationale: rationale }) });
  } catch (e) {}
  credNatForm.busy = false;
  if (resp && resp.ok) {
    const out = await resp.json().catch(() => ({}));
    credNatForm = { entry: "", rationale: "", busy: false };
    credNatMsg = { text: "added — row " + (out.id != null ? out.id : "?") + " · ledger seq " + (out.ledger_seq != null ? out.ledger_seq : "?"), tone: "ok" };
    credNativeLoad(true); // re-read the truth (the worker committed it; the list must show it)
    return;
  }
  const f = resp ? await credWriteFail(resp) : { msg: "network error — the control plane did not answer; the rule was NOT added. Retry." };
  credNatMsg = { text: f.msg, tone: f.elevate ? "dim" : "bad" };
  credRerenderView();
  if (f.elevate && typeof cfgElevateOverlay === "function")
    // the typed entry/rationale survive (the form re-renders FROM STATE) and nothing auto-submits —
    // Add is a deliberate second click after the step-up.
    cfgElevateOverlay("Adding a native rule needs an elevated session.", () => credRerenderView());
}

async function credNativeDelete(id) {
  const rationale = credNatDel.rationale.trim();
  if (!rationale) {
    credNatDel.note = "a rationale is required — taking a mapping back is a decision, not a cleanup";
    credRerenderView();
    return;
  }
  let resp = null;
  try {
    resp = await fetch("/api/v1/credentials/native/rules/" + encodeURIComponent(id), { method: "DELETE", credentials: "same-origin",
      headers: { "Content-Type": "application/json" }, body: JSON.stringify({ rationale: rationale }) });
  } catch (e) {}
  if (resp && resp.ok) {
    const out = await resp.json().catch(() => ({}));
    credNatDel = { id: 0, rationale: "", note: "" };
    credNatMsg = { text: "removed — row " + id + " · ledger seq " + (out.ledger_seq != null ? out.ledger_seq : "?"), tone: "ok" };
    credNativeLoad(true);
    return;
  }
  if (resp && resp.status === 404) {
    // ErrNoSuchNativeRule: the row was gone before this delete — already removed. Converged, said
    // plainly, and the list re-reads so the surface shows the truth rather than a stale row.
    credNatDel = { id: 0, rationale: "", note: "" };
    credNatMsg = { text: "already removed — row " + id + " no longer exists; re-reading the mapping", tone: "dim" };
    credNativeLoad(true);
    return;
  }
  const f = resp ? await credWriteFail(resp) : { msg: "network error — the control plane did not answer; the rule may still exist. Retry." };
  credNatDel.note = f.msg;
  credRerenderView();
  if (f.elevate && typeof cfgElevateOverlay === "function")
    cfgElevateOverlay("Removing a native rule needs an elevated session.", () => credRerenderView());
}

function credNativeDeleteCell(r) {
  const cell = h("td", { class: "cred-actcell" });
  if (credNatDel.id === r.id) {
    const rat = h("input", { id: "credNatDelRat", class: "cred-search m", placeholder: "why is this mapping withdrawn?",
      "aria-label": "Delete rationale for rule " + r.id, value: credNatDel.rationale,
      oninput: e => { credNatDel.rationale = e.target.value; },
      onkeydown: e => { if (e.key === "Enter") { e.preventDefault(); credNativeDelete(r.id); } } });
    const go = h("button", { class: "btn cred-actbtn", type: "button" }, "CONFIRM DELETE");
    go.addEventListener("click", () => credNativeDelete(r.id));
    const no = h("button", { class: "btn cred-actbtn", type: "button" }, "CANCEL");
    no.addEventListener("click", () => { credNatDel = { id: 0, rationale: "", note: "" }; credRerenderView(); });
    cell.append(h("div", { class: "cred-natdel" }, rat, go, no,
      credNatDel.note ? h("div", { class: "desc", style: "flex-basis:100%;font-size:10.5px;color:var(--deviation)" }, credNatDel.note) : null));
    return cell;
  }
  const del = h("button", { class: "btn cred-actbtn", type: "button", "aria-label": "Delete rule " + r.id }, "DELETE");
  del.addEventListener("click", () => { credNatDel = { id: r.id, rationale: "", note: "" }; credRerenderView(false, "credNatDelRat"); });
  cell.append(del);
  return cell;
}

function credNativeSection() {
  const strip = h("div", { class: "cred-strip" });
  strip.append(credSec("LIVE · native mapping", "Operator-authored per-target rules (native-db)"));
  if (credNat.phase === "unloaded") credNativeLoad();
  if (credNat.phase === "unloaded" || credNat.phase === "loading") { strip.append(credErr("Reading the native mapping…")); return strip; }
  if (credNat.phase === "err") {
    if (credNat.status === 403) {
      const eb = h("button", { class: "btn primary", type: "button", id: "credNatElevate" }, "Elevate (admin)");
      eb.addEventListener("click", () => {
        if (typeof cfgElevateOverlay === "function") cfgElevateOverlay("Reading the native mapping needs an elevated session.", credAfterElevate);
      });
      strip.append(h("div", { class: "empty", style: "padding:26px 20px" },
        h("div", { class: "big" }, "Elevated read required"),
        h("div", { class: "desc", style: "max-width:66ch;margin:6px auto 12px" },
          "elevated read required — the rows carry credential references: each entry pairs a target selector with the "
          + "SecretRef reference that unlocks it, so listing them takes trace-read standing (an admin-eligible session, "
          + "or a live admin step-up). References are non-secret by INV-13; resolved values are never served here."),
        eb));
      return strip;
    }
    const retry = h("button", { class: "btn", type: "button", id: "credNatRetry" }, "RETRY");
    retry.addEventListener("click", () => credNativeLoad(true));
    strip.append(h("div", { class: "empty", style: "padding:22px 20px" },
      h("div", { class: "desc", style: "margin-bottom:10px" },
        credNat.timeout ? "The native-mapping read timed out — we do not know its state (this is not the endpoint saying no)."
          : credNat.status === 401 ? "The session is no longer authenticated — sign in again to read the native mapping."
          : "/v1/credentials/native is unavailable on this control plane (fail-closed" + (credNat.status ? " · " + credNat.status : "") + ")."),
      retry));
    return strip;
  }
  const rows = credNat.rows || [];
  if (!rows.length) {
    strip.append(credNone("No native rules", "The native-db mapping holds no operator-authored rows yet — add one below; it lands in "
      + "resolution through the registered native-db source, no restart needed."));
  } else {
    const t = h("table", { class: "tbl" });
    t.append(h("thead", {}, h("tr", {}, ...["ID", "Entry (SecretRefs stay references)", "Rationale", "By", "Created", "Actions"].map(x => h("th", {}, x)))));
    const tb = h("tbody");
    rows.forEach(r => {
      tb.append(h("tr", {},
        h("td", { class: "m", style: "color:var(--ink3)" }, String(r.id != null ? r.id : "—")),
        h("td", { class: "m", style: "color:var(--territory)" }, r.entry || "—"),
        h("td", { style: "color:var(--ink2);font-size:11.5px" }, r.rationale || "—"),
        h("td", { class: "m", style: "color:var(--ink3)" }, r.created_by || "—"),
        h("td", { class: "m", style: "color:var(--ink3)" }, r.created_at ? credRel(r.created_at) : "—"),
        credNativeDeleteCell(r)));
    });
    t.append(tb);
    strip.append(credCard(t));
  }
  strip.append(credNativeAddForm());
  return strip;
}

/* ---- section 2: coverage — "what can TG reach?" ----------------------------------------------- */
function credBar(r, a, u) {
  const total = (r + a + u) || 1;
  const seg = (v, c) => v > 0 ? h("i", { class: "cred-seg", style: "width:" + (v / total * 100).toFixed(2) + "%;background:" + c }) : null;
  return h("div", { class: "cred-bar", title: r + " resolved · " + a + " ambiguous · " + u + " unresolved" },
    seg(r, "var(--match)"), seg(a, "var(--partial)"), seg(u, "var(--deviation)"));
}
function credCovRow(c) {
  return h("div", { class: "cred-cov-row" },
    h("span", { class: "cred-cov-key m", title: c.key || "" }, c.key || "—"),
    h("span", { class: "cred-cov-bar" }, credBar(c.resolved || 0, c.ambiguous || 0, c.unresolved || 0)),
    h("span", { class: "cred-cov-nums m" },
      h("b", { style: "color:var(--match)" }, String(c.resolved || 0)), h("span", { style: "color:var(--ink4)" }, " / "),
      h("span", { style: "color:var(--partial)" }, String(c.ambiguous || 0)), h("span", { style: "color:var(--ink4)" }, " / "),
      h("span", { style: "color:var(--deviation)" }, String(c.unresolved || 0)),
      h("span", { style: "color:var(--ink4)" }, " · " + (c.total || 0))));
}
function credCovGroup(title, rows) {
  const box = h("div", { class: "wrapcard cred-covcard" });
  box.append(h("div", { class: "cred-covhead lbl" }, title));
  if (!rows || !rows.length) { box.append(h("div", { class: "cred-cov-empty desc" }, "no outcomes in window")); return box; }
  rows.forEach(c => box.append(credCovRow(c)));
  box.append(h("div", { class: "cred-covlegend" },
    h("span", {}, h("i", { style: "background:var(--match)" }), "resolved"),
    h("span", {}, h("i", { style: "background:var(--partial)" }), "ambiguous"),
    h("span", {}, h("i", { style: "background:var(--deviation)" }), "unresolved")));
  return box;
}
function credTargetItem(x, ok) {

function credTargetList(title, items, ok) {
  const box = h("div", { class: "wrapcard cred-listcard" });

  items.forEach(x => box.append(credTargetItem(x, ok)));

}
function credCoverageSection() {
  const strip = h("div", { class: "cred-strip" });
  const cov = liveState.credCoverage;
  const win = cov && cov.window_days != null ? h("span", { class: "lbl", style: "font-size:9px" }, "last " + cov.window_days + "d") : null;
  strip.append(credSec("LIVE · coverage", "What can TG reach?", win));
  if (cov === undefined) { strip.append(credErr("Connecting to the credential engine…")); return strip; }
  if (cov === null) { strip.append(credErr("/v1/credentials/coverage is not served by this control plane yet (fail-closed).")); return strip; }

  const hasTally = (cov.by_plane && cov.by_plane.length) || (cov.by_source && cov.by_source.length);
  const hasLists = (cov.recent_resolved && cov.recent_resolved.length) || (cov.recent_refused && cov.recent_refused.length);
  if (!hasTally && !hasLists) {
    strip.append(credNone("No resolutions in the window", "Coverage is derived from real resolution history — once targets resolve or are refused, the frontier appears here."));
    return strip;
  }

    credCovGroup("By source", cov.by_source)));

    credTargetList("Recently resolved", cov.recent_resolved, true),
    credTargetList("Recently refused", cov.recent_refused, false)));
  return strip;
}

/* ---- section 3: resolution log — target filter + limit --------------------------------------- */
let credResQuery = "";
let credResLimit = 50;
let credResLoading = false;
let credResDeb = null;

async function credRunResSearch() {
  credResLoading = true;
  credRerenderView(true);
  const q = credResQuery.trim();
  const qs = "?limit=" + encodeURIComponent(credResLimit) + (q ? ("&target=" + encodeURIComponent(q)) : "");
  try { const p = await liveGet("/v1/credentials/resolutions" + qs); liveState.credResolutions = p.resolutions || []; }
  catch (e) { liveState.credResolutions = null; }
  credResLoading = false;
  credRerenderView(true);
}

/* re-render the whole surface in place (sources/coverage are cached on liveState — cheap), then
   restore focus + caret to the search box, exactly like the knowledge surface's live filter.
   focusId (TG-109): after a state-driven rebuild, put focus on the named control instead — the
   delete-rationale prompt must receive focus when it appears, or a keyboard operator loses their place. */
function credRerenderView(focusSearch, focusId) {
  if ((location.hash || "").replace("#", "") !== "credentials") return;
  const host = document.querySelector(".main .view");
  if (!host) return;
  host.innerHTML = "";
  host.append(views.credentials());
  if (focusSearch) {
    const el = host.querySelector("#credResSearch");
    if (el) { el.focus(); el.setSelectionRange(el.value.length, el.value.length); }
  } else if (focusId) {
    const el = host.querySelector("#" + focusId);
    if (el) el.focus();
  }
}

function credResolutionsSection() {
  const strip = h("div", { class: "cred-strip" });
  const search = h("input", {
    id: "credResSearch", class: "cred-search m", type: "search", "aria-label": "Filter credentials by target", placeholder: "filter by target…",
    value: credResQuery, autocapitalize: "none", spellcheck: "false",
    oninput: e => { credResQuery = e.target.value; clearTimeout(credResDeb); credResDeb = setTimeout(credRunResSearch, 320); },
    onkeydown: e => { if (e.key === "Enter") { e.preventDefault(); clearTimeout(credResDeb); credRunResSearch(); } }
  });
  const note = credResLoading
    ? h("span", { class: "lbl", style: "font-size:9px;color:var(--ink3)" }, "searching…")
    : h("span", { class: "lbl", style: "font-size:9px" }, "newest first · limit " + credResLimit);
  strip.append(h("div", { class: "sec cred-ressec" },
    h("span", { class: "lbl", style: "color:var(--match)" }, "LIVE · resolution log"),
    h("h3", {}, "Recent resolutions"),
    h("span", { class: "rule" }),
    search, note));

  const rs = liveState.credResolutions;
  if (rs === undefined) { strip.append(credErr("Connecting to the credential engine…")); return strip; }
  if (rs === null) { strip.append(credErr("/v1/credentials/resolutions is not served by this control plane yet (fail-closed).")); return strip; }
  if (!rs.length) {
    strip.append(credResQuery.trim()
      ? credNone("No resolutions for “" + credResQuery.trim() + "”", "No audit rows match that target in the recent tail.")
      : credNone("No resolutions recorded", "The resolver has not logged a per-target resolve yet — a quiet log is an idle engine."));
    return strip;
  }

  const t = h("table", { class: "tbl" });
  t.append(h("thead", {}, h("tr", {}, ...["Target", "Plane", "Outcome", "Source", "User", "Scheme", "Key ref", "Resolved"].map(x => h("th", {}, x)))));
  const tb = h("tbody");
  rs.forEach(r => {
    const shadow = r.shadowed && r.shadowed.length ? "also matched: " + r.shadowed.join(", ") : "";
    const srcCell = r.native
      ? h("span", { style: "color:var(--ink3)", title: shadow }, "native")
      : (r.source ? h("span", { style: "color:var(--map)", title: shadow }, r.source) : h("span", { style: "color:var(--ink4)" }, "—"));
    tb.append(h("tr", {},
      h("td", { class: "m terr" }, r.target || "—"),
      h("td", {}, credPlane(r.plane)),
      h("td", {}, credResChip(r.outcome, r.err)),
      h("td", { class: "m" }, srcCell),
      h("td", { class: "m", style: "color:var(--ink2)" }, r.resolved_user || "—"),
      h("td", { class: "m", style: "color:var(--ink2)" }, r.scheme || "—"),
      h("td", { class: "m" }, r.key_ref_scheme ? h("span", { class: "cred-kref" }, r.key_ref_scheme) : h("span", { style: "color:var(--ink4)" }, "—")),
      h("td", { class: "m", style: "color:var(--ink3)" }, credRel(r.created_at))));
  });
  t.append(tb);
  strip.append(credCard(t));
  return strip;
}
