/* ★ #workflows HAD ZERO FOCUSABLE ELEMENTS. Measured live at 1440x900 in both themes: `#view a[href],
   button:not([disabled]), input, select, textarea, [tabindex]:not([tabindex="-1"])` inside #workflows
   returned 0 — and so did [role], [tabindex], button and a[href]. Every one of the 50 run rows, the 10 stage
   headers and the ReAct kid rows was a bare <div> with an onclick. A keyboard or screen-reader operator could
   reach the view and then do NOTHING in it: no run selectable, no stage expandable. WCAG 2.1.1 is Level A.
   wfKeyActivate is shared so the next clickable div added here cannot quietly be mouse-only. */
const wfKeyActivate = e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); e.currentTarget.click(); } };

/* wfListKey is the role=listbox keyboard contract for the governed-run list. Enter/Space activate (unchanged
   from wfKeyActivate, which the other clickable divs still share); ArrowUp/ArrowDown move by one, Home/End to
   the ends. Movement SELECTS as it goes, which is the single-select listbox convention and is what makes the
   detail pane follow the arrows — the whole point of the widget.
   It reads the options from the DOM rather than from wfVisibleRuns() so it can never disagree with what is
   actually on screen after a facet filter. */
function wfListKey(e, list, main){
  if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); e.currentTarget.click(); return; }
  const keys = ['ArrowDown','ArrowUp','Home','End'];
  if (!keys.includes(e.key)) return;
  const opts = Array.from((list || document).querySelectorAll('[role=option]'));
  if (!opts.length) return;
  const cur = opts.indexOf(e.currentTarget);
  let next = cur;
  if (e.key === 'ArrowDown') next = Math.min(opts.length - 1, cur + 1);
  else if (e.key === 'ArrowUp') next = Math.max(0, cur - 1);
  else if (e.key === 'Home') next = 0;
  else if (e.key === 'End') next = opts.length - 1;
  if (next === cur) { e.preventDefault(); return; }
  e.preventDefault();
  const key = opts[next].getAttribute('data-wf-key') || '';
  const id = key.replace(/^run:/, '');
  if (id) wfSelect(id, list, main);
  // Re-resolve the node AFTER the re-render selection triggers: the original element is detached by then, so
  // focusing it would silently drop focus to <body> — the defect this file already fixed once.
  const again = (list || document).querySelector('[data-wf-key="' + (window.CSS && CSS.escape ? CSS.escape(key) : key) + '"]');
  (again || opts[next]).focus();
}
/* ★ AND THEN THE LIVE POLL THREW THAT KEYBOARD OPERATOR OFF THE PAGE. Arming the view was only half the job.
   Both panes re-render by replaceChildren(), which DETACHES the focused node — so measured live: 15 of 15
   stage toggles moved focus to <body> on activation, every run activation did the same, and worst of all it
   happened with NO USER INPUT AT ALL, because the 1s liveness tick and the SSE-driven refresh re-render on
   their own. A screen-reader operator who tabbed 34 times to reach a run lost the reading cursor to the top
   of the document within 20 seconds of arriving, forever, with nothing they did causing it.

   A focus-destroying re-render is not a smaller bug than an unfocusable div: it is the same bug arriving on a
   timer. So the identity that survives a re-render is explicit — every focusable node in this view carries a
   stable data-wf-key — and both render entry points go through this wrapper. If the keyed node is gone
   (facet changed, run vanished) focus lands on the pane itself, which is a place; it never lands on <body>,
   which is not. */
function wfKeyed(el, key, attrs){ return Object.assign(attrs || {}, {'data-wf-key': key}); }
function wfPreserveFocus(container, render){
  const a = document.activeElement;
  const key = (container && a && a !== document.body && container.contains(a)) ? a.getAttribute('data-wf-key') : null;
  const scrolled = key ? null : undefined;
  render();
  if(key === null) return;
  let want = container.querySelector('[data-wf-key="' + (window.CSS && CSS.escape ? CSS.escape(key) : key) + '"]');
  if(!want) want = container.querySelector('[tabindex="0"],[role=option],[role=button]');
  if(want && typeof want.focus === 'function'){ want.focus({preventScroll:true}); return; }
  // last resort: the pane itself is a place to stand. Never <body>.
  if(!container.hasAttribute('tabindex')) container.setAttribute('tabindex','-1');
  container.focus({preventScroll:true});
}

function wfSelect(id, list, main){
  const changed = wfSelId !== id;
  wfSelId = id;
  if(wfIsLive() && typeof wfOnSelect === 'function') wfOnSelect(id); // async: fetch+map the walk, then re-render
  if(list) wfRenderList(list, main);
  if(main) wfRenderMain(main);
  if(changed){
    const r = wfAllRuns().find(x => x.id === id);
    if(r && typeof announce === 'function')
      announce('Run ' + r.ref + ' selected — ' + r.title + ' on ' + r.host + ', ' +
               (bandName[r.band] || r.band) + ', ' + (r.nodes ? r.nodes.length : 0) + ' stages shown.');
  }
}

function wfDurF(ms){
  if(ms == null) return '—';
  if(ms < 1000) return ms + 'ms';
  if(ms < 60000) return (ms/1000).toFixed(1) + 's';
  return Math.floor(ms/60000) + 'm ' + String(Math.round(ms%60000/1000)).padStart(2,'0') + 's';
}
function wfElapsedF(sec){
  if(sec < 60) return sec + 's';
  if(sec < 3600) return Math.floor(sec/60) + 'm ' + String(sec%60).padStart(2,'0') + 's';
  return Math.floor(sec/3600) + 'h ' + String(Math.floor(sec%3600/60)).padStart(2,'0') + 'm';
}
/* A live run's elapsed is DERIVED from its real start instant, never counted up in the browser — a
   client-side counter is a simulation of a fact the spine already owns, and it drifts. Fixture runs keep
   their simulated counter; a run with neither answers "—" rather than "0s", because "we do not know how
   long" and "it just started" are different things to tell someone deciding whether to intervene. */
function wfElapsedText(r){
  if(!r) return '—';
  if(r.startedMs) return wfElapsedF(Math.max(0, Math.round((Date.now() - r.startedMs)/1000)));
  return (r.elapsed == null) ? '—' : wfElapsedF(r.elapsed);
}

/* The armed-revert chip (spec/029 T-029-5, REQ-2906): the commit_confirm window's durable state,
   rendered only when the walk carries one — the dead-man's switch made visible. Glyph/color per
   state family: armed counts down, confirmed is the clean end, a fired/failed revert is loud. */
function wfCommitConfirmChip(cc){
  if(!cc || !cc.state) return null;
  const m = {
    armed:               ['ARMED',        '--pause',     '◔'],
    confirmed:           ['CONFIRMED',    '--match',     '✓'],
    aborted:             ['ABORTED',      '--ink3',      '∅'],
    elapsed_unconfirmed: ['ELAPSED',      '--partial',   '◐'],
    held_unverifiable:   ['HELD',         '--hazard',    '‖'],
    reverted:            ['REVERTED',     '--deviation', '↩'],
    revert_failed:       ['REVERT FAILED','--deviation', '✕'],
  }[cc.state] || [String(cc.state).toUpperCase(), '--ink4', '?'];
  const tip = 'commit-confirmed window'
    + (cc.op_class ? ' · ' + cc.op_class : '')
    + (cc.deadline_at ? ' · deadline ' + cc.deadline_at : '')
    + (cc.detail ? '\n' + cc.detail : '')
    + (cc.inverse_action_id ? '\ninverse: ' + cc.inverse_action_id : '');
  return h('span',{class:'wf-chip', style:'color:var(' + m[1] + ')', title: tip}, h('span',{class:'g'}, m[2]), m[0]);
}
function wfRunChip(r){
  if(r.verdict) return wfVerdictChip(r.verdict);
  const m = { running:['RUNNING','--map','▶'], waiting:['AWAITING APPROVAL','--pause','‖'],
              floored:['FLOORED','--hazard','◆'], failed:['FAILED','--deviation','✕'],
              done:['DONE','--ink3','✓'] }[r.status] || ['—','--ink4',''];
  const cls = 'wf-chip' + (r.status === 'floored' ? ' wf-mut' : '');
  return h('span',{class:cls, style: r.status === 'floored' ? null : 'color:var(' + m[1] + ')'}, h('span',{class:'g'}, m[2]), m[0]);

function wfRenderListInner(list, main){
  list.replaceChildren();
  const runs = wfVisibleRuns();
  list.append(h('div',{class:'wf-lhead'},
    'Runs · ' + runs.length + '/' + wfAllRuns().length,
    wfIsLive()
      ? h('span',{class:'wf-fix', style:'color:var(--match)', title:'Live — assembled from the governance spine (/v1/sessions)'}, 'live')
      // On a live shell a fixture fallback is named by the GLOBAL fixture banner (fixtureNote), so the
      // inline chip renders only on the unauthenticated design preview — one label, one place.
      : ((typeof liveState!=="undefined" && liveState && liveState.on)
        ? null
        : h('span',{class:'wf-fix', title:'Design preview — representative fixtures, not live telemetry'}, 'fixture data'))));
  runs.forEach(r => {
    list.append(h('div',{class:'wf-run' + (r.id === wfSelId ? ' sel' : ''),
        // ★ THE COMPOSITE-WIDGET CONTRACT, not just the role. This declared role=listbox with 50 role=option

        // Home/End. A screen reader announced "listbox, 50 items" and the arrow keys did nothing, which is a
        // worse position than an unlabelled list — the operator is told a navigation model that does not exist.

        // tabindex is now ROVING: exactly one option is tabbable (the selected one), so the list is ONE tab

        'aria-selected': String(r.id === wfSelId), 'data-wf-key': 'run:' + r.id,
        onclick: () => wfSelect(r.id, list, main), onkeydown: e => wfListKey(e, list, main)},
      h('span',{class:'wf-stripe', style:'background:var(' + wfBandTok[r.band] + ')'}),

        h('span',{class:'wf-bandsm', style:'color:var(' + wfBandTok[r.band] + ')'}, bandName[r.band]))));

function wfRenderMain(main){
  if(!main) return; // a detached/rapidly-replaced node's onclick can pass a null container — no-op, never throw
  return wfPreserveFocus(main, () => wfRenderMainInner(main));
}
function wfRenderMainInner(main){
  const st = main.scrollTop;
  main.replaceChildren();
  const r = wfAllRuns().find(x => x.id === wfSelId);
  if(!r){ main.append(h('div',{class:'wf-empty'}, 'Select a run.')); return; }
  main.append(wfHead(r));
  // Live runs lazy-load their walk from /v1/sessions/{ref}; reflect that honestly instead of an empty timeline.
  if(r._err !== undefined && r._err !== null){
    // NAME THE CAUSE, THEN THE ACTION THAT WORKS. This read used to answer every failure with "The trace
    // endpoint is unavailable (403) — retry.", which was wrong twice over: the endpoint was fine (the session
    // simply lacked the elevated trace-read role), and retrying is precisely the thing that cannot help — the
    // pre-!683 client took that instruction literally and issued 28,094 rejected requests at ~75/s. An
    // operator followed the same advice by hand and lost time to it.
    //
    // 403 on this GET means one thing on this surface: the session does not carry trace-read. Signing in
    // again re-derives it from the directory, which is the fix when a control-plane restart dropped the grant
    // (see REQ-527) and a truthful statement of what is missing when the account never had it.
    main.append(h('div',{class:'wf-empty'},
      r._err === 403 ? 'This session does not carry the trace-read role. Sign in again to refresh it.'
      : r._err === 404 ? 'That session was not found (unknown or unauthorized reference).'
      : 'The trace endpoint answered ' + r._err + '. Retry, or check the control plane if it persists.'));
  } else if(r._loading){
    main.append(h('div',{class:'wf-empty'}, 'Loading the governed walk…'));
  } else if(wfIsLive() && (!r.nodes || !r.nodes.length)){
    main.append(h('div',{class:'wf-empty'}, 'This session recorded no decision boundaries yet.'));
  } else {
    main.append(wfTimeline(r));
  }
  main.scrollTop = st;
}

    wfIcEl(n.t),

    n.band ? wfBandChip(n.band) : null,
    n.hash ? h('span',{class:'wf-hash'}, n.hash) : null,
    n.gate ? wfGateChip(n.gate) : null,
    n.t === 'verify' && n.verdict ? wfVerdictChip(n.verdict) : null,
    n.st === 'waiting' ? h('span',{class:'wf-chip', style:'color:var(--pause)'}, h('span',{class:'g'},'‖'), 'PAUSED') : null,
    n.retry ? h('span',{class:'wf-retry'}, 'retry ×' + n.retry) : null,
    h('span',{class:'wf-sp'}),
    n.dur != null ? h('span',{class:'wf-db'},
      h('span',{class:'wf-dbf', style:'width:' + Math.max(4, Math.round(100 * Math.sqrt(n.dur / maxDur))) + '%'})) : null,
    h('span',{class:'wf-durv'}, n.st === 'active' ? '···' : wfDurF(n.dur)),
    h('span',{class:'wf-ts'}, n.ts || ''),
    // aria-expanded already carries this state; the glyph is the SIGHTED channel for the same fact and
    // was being spoken as "black right-pointing small triangle" at the end of every stage header's name.
    h('span',{class:'wf-chev','aria-hidden':'true'}, hasPay ? (open ? '▾' : '▸') : ''));
  return h('div',{class:'wf-node st-' + n.st},
    h('div',{class:'wf-rail' + (n.st === 'active' ? ' wf-flow' : '')}, h('span',{class:dotCls})),
    h('div',{class:'wf-card'},
      hdr,
      open ? wfPayEl(n) : null,
      open ? wfCmpEl(n) : null,
      n.kids ? h('div',{class:'wf-kids'}, n.kids.map((k, ki) => wfKidEls(r, i, k, ki)).flat()) : null,
      open && n.cta ? h('div',{class:'wf-cta'},
        h('button',{class:'wf-btn', disabled:true}, n.cta),
        h('span',{class:'wf-gnote'}, postureClaim().state==='off' ? postureClaim().label + ' — read-only preview'
          : postureClaim().state==='on' ? postureClaim().label + ' — approved actions execute'
          : postureClaim().label)) : null));
}

/* ---------------- liveness: 1s clock + advancing run (reduceMotion-guarded) ---------------- */
function wfStartClock(root, main){
  wfTimer = setInterval(() => {
    if(!document.contains(root)){ clearInterval(wfTimer); wfTimer = null; return; }
    wfTickN++;
    /* ★ THE TICKER ITERATED THE FIXTURES WHILE THE VIEW RENDERED THE LIVE RUNS. wfAllRuns() returns
       wfRunsSrc||WF_RUNS, so on a live estate the rows on screen come from wfRunsSrc while this lookup
       searched WF_RUNS — no id ever matched, nothing was ever updated, and every live run displayed
       "elapsed 0s" permanently. The clock has to read the same source the renderer does.
       Only fixture runs are INCREMENTED; a live run's elapsed is derived from its real start instant, so
       counting it here would be simulating a fact the spine already knows. */
    wfAllRuns().forEach(r => { if(r.live && !r.startedMs && r.elapsed != null) r.elapsed++; });
    $$('[data-wf-clock]').forEach(el => {
      const r = wfAllRuns().find(x => x.id === el.getAttribute('data-wf-clock'));
      if(r) el.textContent = wfElapsedText(r);
    });
    if(!reduceMotion && wfTickN % 6 === 0) wfAdvance(main);
  }, 1000);
}
function wfAdvance(main){
  if(wfIsLive()) return; // live runs update from the real SSE step channel, never a client-side simulation
  const r = WF_RUNS.find(x => x.live && !x._advDone);
  if(!r) return;
  const ag = r.nodes.find(n => n.t === 'agent' && n.st === 'active');
  if(!ag || !ag.kids) return;
  const act = ag.kids.find(k => k.st === 'active');
  if(act){ act.st = 'done'; if(act.dur == null) act.dur = 300 + Math.round(Math.random() * 1400); }
  if(ag.kids.length >= 9){ r._advDone = true; }
  else {
    const nx = WF_TOOL_POOL[wfPoolI % WF_TOOL_POOL.length]; wfPoolI++;
    ag.kids.push({tool: nx.tool, q: nx.q, st:'active', dur: null, res: nx.res, retry: 0});
  }
  if(wfSelId === r.id) wfRenderMain(main);
}
