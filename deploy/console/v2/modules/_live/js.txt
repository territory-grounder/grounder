/* ★ EVERY LIVE READ IS BOUNDED. liveAdopt awaits liveGet 26 times in sequence, and this had no timeout and
   no AbortController: a single endpoint that was merely SLOW — not failing, not returning an error, just
   hanging — stalled the entire adoption behind it. Nothing downstream ran, so the console sat on its skeleton
   with no error, no spinner and no way for the operator to tell the difference between "the estate is quiet"
   and "the console stopped reading". A read that never resolves is the one failure the fail-closed rendering
   work could not catch, because it never reaches a branch.
   A TIMEOUT converts that into an ordinary rejection, which every call site already handles (each is wrapped
   in try/catch and degrades to an honest empty state). AbortSignal also releases the socket instead of
   leaving 26 of them parked.
   The budget is per-request, deliberately generous, and NOT configurable from the page: it exists to bound a
   pathological case, not to tune performance. */
const LIVE_READ_TIMEOUT_MS = 12000;
async function liveGet(path, opts){
  const budget = (opts && opts.timeoutMs) || LIVE_READ_TIMEOUT_MS;
  const ctl = (typeof AbortController === "function") ? new AbortController() : null;
  let timer = null;
  if(ctl) timer = setTimeout(() => { try{ ctl.abort(); }catch(e){} }, budget);
  try{
    const r = await fetch("/api"+path, Object.assign(
      { credentials:"same-origin" }, ctl ? { signal: ctl.signal } : {}));
    if(!r.ok) throw Object.assign(new Error(path+" -> "+r.status), { status:r.status });
    return await r.json();
  }catch(e){
    // A timeout must be DISTINGUISHABLE from a server error at the call site: an aborted read means "we do
    // not know", never "the endpoint said no".
    if(e && (e.name === "AbortError" || e.code === 20)){
      throw Object.assign(new Error(path+" -> timed out after "+budget+"ms"), { timeout:true, path });
    }
    throw e;
  }finally{ if(timer) clearTimeout(timer); }
}

/* loginFailureText maps a login response status to a sentence that names the CAUSE and the ACTION. Split
   out so it is testable without a browser session and so the closed set of statuses is visible in one
   place — see e2e/login-error-honesty.mjs, which asserts over that set rather than over one sample. */
function loginFailureText(status){
  if(status===401 || status===403) return "unauthenticated — check the operator name and password";
  if(status===404) return "sign-in is UNAVAILABLE on this control plane (404): the session route is not "+
    "registered, which happens when the grounder could not resolve its session key at boot. This is not "+
    "your password. An operator with host access should check the grounder log for \"browser sessions "+
    "disabled\" and restart it.";
  if(status>=500) return "the control plane is failing this request ("+status+") — sign-in cannot complete. "+
    "This is not your password. Retry shortly; if it persists, check the grounder.";
  return "sign-in failed ("+status+") — this is not necessarily your password; report the status shown.";
}

async function liveLogin(e){
  if(e) e.preventDefault();
  const op=$("#agOp"), tok=$("#agTok");
  const name=(op&&op.value||"").trim(), token=(tok&&tok.value)||"";
  if(!name||!token){ agSpine("err"); agErr("username and password are required"); return; }
  agBusy(true); agSpine("verifying"); agErr("");
  let r=null;
  try{
    r = await fetch("/api/v1/session",{ method:"POST", credentials:"same-origin",
      headers:{ "X-TG-Operator":name, "Authorization":"Bearer "+token } });
  }catch(ex){ agBusy(false); agSpine("err"); agErr("network error — retry"); return; }
  /* ★ EVERY NON-429 FAILURE READ AS "YOUR CREDENTIALS ARE WRONG", INCLUDING A CONTROL PLANE THAT IS DOWN.
     I hit this myself on 2026-07-29. A transient OpenBao stall at boot made the grounder disable browser
     sessions, which UN-REGISTERS /v1/session — so POST returned 404 while every other route returned 401.
     This line rendered that 404 as "unauthenticated — check the operator name and token", and I spent the
     first minutes of a live outage re-checking a credential that was never the problem.

     The status is the diagnosis and it was being thrown away. 401 is the ONLY one that means what the old
     text said. 404 means the login route is not registered — a server-side condition an operator cannot fix
     by retyping anything, and the one that actually happened. 5xx means the control plane is failing. On a
     console that owns the KILL switch and the approval queue, telling the operator to check their password
     during an outage is the most expensive wrong sentence available.

     Each branch says WHAT is wrong and WHAT ACTION WORKS, because "retry" is useless advice for a 404 and
     the absence of that distinction is the defect. The status is included verbatim so a screenshot is
     diagnosable without a log. */
  if(r.status===429){ agBusy(false); agSpine("err"); agErr("rate limited — wait a minute and retry"); return; }
  if(!r.ok){
    agBusy(false); agSpine("err");
    agErr(loginFailureText(r.status));
    return;
  }
  agSpine("ok");
  await liveAdopt();
}

/* liveInjectEstateDepth wires the Estate Depth inspector (#estatedepth) to REAL data by replacing the
   fixture globals its render reads (ESTATE.nodes/edges, HOSTS, ALERTS) with the live estate graph
   (/v1/estate) and accepted-alert window (/v1/alerts) — the LEDGER.push pattern. The estate2 render
   (estateNodeDetail/est2View) is consumed UNCHANGED: node list, dependencies, blast radius, node metadata,
   and active-alerts become real. Node health is DERIVED honestly from the alerts on that node (a crit alert
   → crit, a warn → warn, else ok) since the estate graph carries no health field. Golden signals + the
   incident master-clock stay their labeled "fixture" (they need a per-node prometheus time-series that has
   no endpoint yet). Sessions-on-node stays empty until the session list carries host. Called once per adopt,
   before the post-adopt re-render, so a direct land on #estatedepth shows real data. */
function liveEstSev(s){ s=String(s||"").toLowerCase(); return s.startsWith("crit")?2:(s.startsWith("warn")?1:0); }
function liveEstSite(name){ const m=String(name||"").match(/^([a-z]{2}[a-z]*\d{2})/i); return m?m[1]:String(name||"").slice(0,7); }
function liveEstTime(t){ if(!t) return ""; const d=new Date(t); return isNaN(d)?"":d.toISOString().slice(11,19); }
function liveInjectEstateDepth(){
  if(typeof ESTATE==="undefined") return; // estate2 module not loaded
  // worst live-alert severity per host → the node's health chip + list sort
  //
  // ★ A FAILED ALERT READ IS NOT A HEALTHY ESTATE. The fetch layer already draws the distinction this
  // function used to erase: liveState.alerts is NULL when /v1/alerts failed and [] when it genuinely
  // returned an empty window. `Array.isArray(x) ? x : []` collapses both to "no alerts", every host then
  // falls through `||"ok"`, and the console paints the whole estate green off a read that never landed —
  // in the reassuring direction, on the lane an operator uses during an incident.
  //
  // The third state is "unknown" and it must survive all the way to the chip. Note this is NOT the same
  // fact as an empty estate: "nothing is unhealthy" and "we could not find out" differ exactly when it
  // matters most.
  const alertsUnknown = liveState.alerts == null;
  const sevByHost={};
  (Array.isArray(liveState.alerts)?liveState.alerts:[]).forEach(a=>{
    if(!a || !a.host) return; const r=liveEstSev(a.severity||a.sev);
    if(!(a.host in sevByHost) || r>sevByHost[a.host]) sevByHost[a.host]=r; });
  const health=hn=> alertsUnknown ? "unknown" : (({2:"crit",1:"warn"})[sevByHost[hn]]||"ok");
  const est=liveState.estate;
  if(est && Array.isArray(est.nodes)){
    // /v1/estate models the same host as MULTIPLE node types — a network-discovered 'host' AND its proxmox
    // 'lxc'/'vm'/'pve_node' (name+type is unique, but ~150 names appear twice). The inspector shows ONE row
    // per host, so dedupe by name, preferring the specific virtualization type over the generic 'host'.
    const byName={};
    est.nodes.forEach(n=>{ const c=byName[n.name];
      if(!c || (c.type==="host" && n.type!=="host")) byName[n.name]={name:n.name,type:n.type}; });
    const uniq=Object.keys(byName).map(k=>byName[k]);
    ESTATE.nodes = uniq.map(n=>({ id:n.name, health:health(n.name), r:n.type, type:n.type }));
    ESTATE.edges = (est.edges||[]).map(e=>[ e.from, e.to, (e.confidence==null?1:e.confidence) ]);
    // The Estate Depth inspector keys on THIS flag, not on liveState.estate: "the fetch returned something"
    // and "the fixture globals were actually replaced" are different facts, and rendering fixture nodes
    // unlabelled on a live shell is what the gap between them cost. One writer: set only here, where the
    // replacement genuinely happened.
    liveState.estateInjected = true;
    if(typeof HOSTS!=="undefined"){
      Object.keys(HOSTS).forEach(k=>delete HOSTS[k]);
      uniq.forEach(n=>{ HOSTS[n.name]={ site:liveEstSite(n.name), role:n.type, health:health(n.name) }; });
    }
  }
  if(typeof ALERTS!=="undefined" && Array.isArray(liveState.alerts)){
    const sevName=s=>({2:"crit",1:"warn"})[liveEstSev(s)]||"info";
    const mapped=liveState.alerts.map(a=>({
      id:a.external_ref||a.source_id||"", src:a.source_type||a.src||"", host:a.host||"",
      rule:a.alert_rule||a.rule||"", sev:sevName(a.severity||a.sev),
      ts:a.ts||liveEstTime(a.observed_at||a.received_at), state:a.state||"active",
      session:a.workflow_id||null }));
    ALERTS.length=0; ALERTS.push(...mapped);
  }
}

// ★ THE CONSOLE FROZE AFTER BOOT WHILE TELLING THE OPERATOR IT WAS LIVE.
// Measured over two independent sessions: 29 requests to /api/ during login, then ZERO — for a 90-second
// stationary window AND across #alerts -> #command -> #alerts -> #governance -> #logs. Leaving a view and
// coming back forced no refetch (reqsDuringRenav: 0). The SSE stream opened once and carried POSTURE only.
// Meanwhile the header's chain-seq advanced live (6004 -> 6007 while the page sat open) and the connection
// chip read "SSE LIVE" at every sample. So the one indicator an operator trusts was actively reassuring them
// the data was current while every list below it was a snapshot frozen at login. During an incident that is
// the worst possible failure: the console looks alive and shows the past.
// liveAdopt now takes `refresh` so the same reads can run again. It already ends by re-rendering the current
// view, and liveStream() is idempotent, so the only things that must not repeat are the boot-time UI and the
// login overlay.
async function liveAdopt(refresh){
  // ★ liveAttempted SEPARATES AN OPERATOR CONSOLE FROM A DESIGN PREVIEW. Every fixture global in this
  // bundle (LEDGER, ESTATE, HOSTS, ALERTS) exists so the page can be opened from disk and look like
  // something. On a served console those fixtures must NEVER reach the screen — not while loading, not on a
  // failed read. Measured 2026-07-29 with /v1/ledger forced to fail: #ledger rendered its 8 design rows into
  // a surface captioned "Tamper-evident record · SHA-256 chain verified", two of them carrying a REAL
  // PERSON'S NAME on an approve and a kill-switch.arm they never performed. This flag is what the views
  // check before they are willing to show a fixture.
  liveState.attempted = true;
  if(!refresh){ agBusy(true); agSpine("verifying"); agErr(""); }
  let who;
  try { who = await liveGet("/v1/whoami"); }
  catch(e){
    // On BOOT a failed whoami means "sign in". On a REFRESH it must not evict a working session on one
    // transient error — mark the data stale, leave the page alone, and let the next tick try again.
    if(refresh){ liveState.refreshFailed = true; liveFreshness(); return; }
    liveLoginOverlay(); return;
  }
  liveState.refreshFailed = false;
  liveState.on = true;
  liveState.operator = String(who.source||"").replace(/^operator:/,"");
  liveState.mode = who.mode ? String(who.mode) : null;
  liveState.mayActuate = !!who.may_actuate;
  liveState.postureStale = !!who.posture_stale;

  // Reveal the console now — identity is verified. Every fetch below runs AFTER this, never before.
  revealConsole();

  // posture bar: identity + posture become live
  const whoEl=$("#whoName"); if(whoEl && liveState.operator) whoEl.textContent = liveState.operator;
  const conn=$("#connTxt"); if(conn) conn.textContent = "API LIVE";
  // The actuation posture reported here is the WORKER's live mode + may_actuate (not the read-only
  // grounder's gate); when it is stale/unpublished the chip shows "unverified", never a confident OFF.
  setBreaker(liveState.mode, liveState.mayActuate, liveState.postureStale, who.posture_source);

  // the preview flag tells the truth per-state
  $$(".preview-flag").forEach(el=>{
    el.innerHTML=""; el.append(h("i",{style:"background:var(--match)"}),
      "LIVE · "+(liveState.operator||"operator")+" · read-only session");
    el.style.borderStyle="solid";
  });
  const foot=$(".rail .foot div:last-child");
  if(foot) foot.innerHTML="Posture, ledger &amp; sessions are <b>live</b> from the control plane.<br>If any view falls back to fixture data, the banner above it names it.";

  /* live ACTIONS — the sealed ActionManifests, replacing five INVENTED incidents that were rendered against
     REAL estate hostnames ("Repeated auth failures -> ASA dc1demo-fw01") while 109 genuine manifests sat
     unread in action_manifest. Each ribbon is mapped onto the SAME shape ribbonRow() already renders, so the
     design is untouched — only the data changes from invented to stored. */
  try{
    const page = await liveGet("/v1/actions?limit=50");
    const BAND = {AUTO:"auto", AUTO_NOTICE:"notice", POLL_PAUSE:"pause"};
    liveState.actions = (page.actions||[]).map(a=>({
      id: a.action_id.slice(0,8),
      ref: a.plan_hash ? "plan:"+a.plan_hash.slice(0,8) : a.action_id.slice(0,8),
      // The TITLE is the sealed manifest read back, never a narrative: "start-service nginx.service".
      title: [a.op_class, a.params && (a.params.unit||a.params.container||a.params.name)].filter(Boolean).join(" "),
      host: a.target, band: BAND[a.band]||"pause",
      // status is DERIVED from the stages that actually happened, in the order they happen.
      // Read the field that says WHY it never ran. approval_choice is obsolete | timeout | approved (or "");
      // only the last of those is on the path to execution, and none of the three means "executing".
      status: a.verdict==="deviation" ? "deviation" : a.verified ? "verified"
            : a.executed ? "executing"
            : a.approval_choice==="obsolete" ? "obsolete"
            : a.approval_choice==="timeout" ? "timeout"
            : a.approved ? "approved"
            : a.band==="POLL_PAUSE" ? "awaiting" : "ready",
      approval_choice: a.approval_choice || "",
      verdict: a.verdict||"",
      // risk is the stored CATEGORY (no score exists — see axes()); conf is the real scalar or absent.
      risk: a.risk_level||"", conf: a.has_confidence ? a.confidence : undefined,
      age: liveAge(a.sealed_at), sealed_at: a.sealed_at, reversible: !!a.reversible,
      exec_class: a.op_class||"none", floored: false,
      action: {id: a.action_id},
      stage: {classified:a.classified?1:0, predicted:a.predicted?1:0, approved:a.approved?1:0,
              executed:a.executed?1:0, verified:a.verified?1:0},
    }));
    liveState.actionCounts = page.counts||null;
  }catch(e){ liveState.actions = null; liveState.actionCounts = null; }

  /* ledgerTime renders governance_ledger.created_at, which the read projected as "" until the column was
     added to the SELECT (core/db/ledger.go). Two rules the surface must keep:
       - a MISSING or ZERO time renders blank, never an epoch. An entry from the in-memory ledger carries
         the zero time; printing "01/01/0001" or "1970" would assert a fact the row does not have.
       - the time is NOT covered by the SHA-256 chain (see audit.LedgerEntry.CreatedAt). This panel's
         "chain verified" chip speaks for the hashed fields only, so the column stays visually quiet
         (ink3, same as seq) rather than reading as verified evidence. */
  /* ★ ONE EVENT, TWO CLOCK TIMES, NEITHER LABELLED. This rendered created_at in the VIEWER'S browser
     timezone while #command and #alerts render UTC, both unlabelled and with no date column here at all —
     so the same action (957f5d4d) appeared at two different times on two views with nothing on screen to
     reconcile them, and a ledger row from yesterday was indistinguishable from one an hour ago. A
     hash-chained audit record whose timestamp cannot be compared across surfaces is not evidence.
     One zone (UTC, the zone the control plane records in), stated in the column header, plus the date. */
  const ledgerTime = iso => {
    if(!iso) return "";
    const d = new Date(iso);
    if(isNaN(d.getTime()) || d.getUTCFullYear() < 2000) return "";
    return d.toISOString().slice(0,19).replace("T"," ");
  };

  // live ledger
  try{
    const page = await liveGet("/v1/ledger");
    if(Array.isArray(page.entries)){
      liveState.ledgerState = "ok";
      liveState.ledger = page.entries.slice(-40).reverse().map(e=>({
        seq:e.seq, ts:ledgerTime(e.created_at), actor: e.decision&&e.decision.startsWith("human")?"human":"system",
        kind:(e.decision||"")+(e.withheld?" (withheld)":""), ref:e.reason||"", action:(e.action_id||"").slice(0,8),
        hash:(e.hash||"").slice(0,4)+"…"+(e.hash||"").slice(-4), prev:(e.prev_hash||"").slice(0,4)+"…"+(e.prev_hash||"").slice(-4)}));
      // Replace UNCONDITIONALLY. The old guard `if(liveState.ledger.length)` kept the 8 design rows
      // whenever the real ledger came back empty — a second, quieter route to the same fabrication.
      LEDGER.length=0; LEDGER.push(...liveState.ledger);
      /* ★ THE CONTROL PLANE'S OWN CONFIGURATION HEALTH WAS FETCHED AND THEN DISCARDED.
         `config:gap-at-boot` is appended when — and only when — the worker finds a configuration gap at
         startup: carve-outs that do not cover allowlisted guests, an evidence-reader domain armed with no
         sanctioned principals (every non-TG actor there reads attributed-suspicious), a PVE TLS flag
         disagreement. Measured live: /v1/ledger returns 200 entries containing ELEVEN such rows, and the
         console renders `entries.slice(-40)` — which held NONE of them. The finding reached the spine, reached
         the browser, and then fell off the end of a window sized for recent activity.
         So the operator could not learn that their own control plane was misconfigured from any surface, while
         the answer sat in memory. Kept here, out of the full page, before the window is applied. */
      liveState.configGap = null;
      for(const e of page.entries){
        if(e && /^config:gap/.test(e.decision||"")) liveState.configGap = e;   // last wins = newest
      }
      const chip=$("#chainSeq"); if(chip && liveState.ledger.length) chip.textContent=String(liveState.ledger[0].seq);
    }
  }catch(e){
    // The old comment here claimed the ledger "stays fixture-labeled". It did not: the view rendered the
    // fixture rows with no label at all, under a "SHA-256 chain verified" chip. A read that failed is a
    // fact the operator needs; it is never 8 plausible rows.
    liveState.ledgerState = "failed";
    LEDGER.length = 0;
  }

  // live alerts (REQ-510) — the accepted-envelope window
  try{
    const page = await liveGet("/v1/alerts?limit=50");
    liveState.alerts = page.alerts||[]; liveState.alertsState = "ok";
    /* counts is the POPULATION behind the page. The badge previously rendered liveState.alerts.length,
       which is the fetch limit — it read "50" whether the estate had emitted 50 alerts or 5,000. */
    liveState.alertCounts = page.counts||null;
  }catch(e){ liveState.alerts = null; liveState.alertCounts = null; liveState.alertsState = "failed"; }

  // live governance posture (REQ-511), secret references (REQ-512), module capabilities
  try{ liveState.governance = await liveGet("/v1/governance"); }catch(e){ liveState.governance=null; }
  try{ const p=await liveGet("/v1/secrets"); liveState.secretRefs=p.refs||[]; liveState.sealed=p.sealed||[]; }catch(e){ liveState.secretRefs=null; liveState.sealed=null; }
  // refresh the nav badge from the REAL missing count (retiring the fixture SEC_REFS count).
  try{ const m=(liveState.secretRefs||[]).filter(r=>!r.resolved).length; $$('[data-badge="secretsMissing"]').forEach(el=>{ el.textContent=String(m); el.style.display=m?"":"none"; }); }catch(e){}
  try{ liveState.caps = await liveGet("/v1/capabilities"); }catch(e){ liveState.caps=null; }

  // live estate graph (REQ-516) — the worker's published causal graph
  try{ liveState.estate = await liveGet("/v1/estate"); liveState.estateState = "ok"; }
  catch(e){
    // Measured live with /v1/estate failing: the inspector kept the bundle's SEVEN design nodes and
    // presented them as the estate, unlabelled, against a real estate of 367 entities and 390 edges.
    // A wrong graph is worse than no graph — an operator reasons about blast radius from it.
    liveState.estate=null; liveState.estateState="failed";

/* ★ STALENESS MUST NEVER BE SILENT. Even with the loop below, a refresh can fail; when it does the operator
   has to be able to see that what they are reading is old. This writes the age of the data next to the
   connection chip, and says so plainly once it is older than two intervals. */
const LIVE_REFRESH_MS = 25000;
function liveFreshness(){
  const el = $("#liveFresh"); if(!el) return;
  const t = liveState.lastRefresh;
  if(!t){ el.textContent = ""; return; }
  const age = Math.round((Date.now() - t) / 1000);
  const stale = liveState.refreshFailed || (age * 1000) > LIVE_REFRESH_MS * 2;
  el.textContent = stale ? ("data " + age + "s old") : ("updated " + age + "s ago");
  el.style.color = stale ? "var(--deviation)" : "var(--ink4)";
  el.title = stale
    ? "The last refresh did not complete. What you are reading is " + age + " seconds old — do not read it as current."
    : "Lists re-read every " + Math.round(LIVE_REFRESH_MS/1000) + "s while this tab is visible.";
}

/* The loop itself. Three guards, each for a reason that showed up in measurement:
   - visibilityState: a background tab must not poll the control plane every 25s forever;
   - an OPEN overlay is skipped, because re-rendering underneath a focused dialog yanks focus out of it —
     the exact defect the overlay work fixed this morning;
   - a refresh already in flight is not started twice on a slow link. */
let liveRefreshTimer = null, liveRefreshInFlight = false;
async function liveRefresh(){
  if(document.visibilityState !== "visible") return;
  if(liveRefreshInFlight) return;
  if(document.querySelector("#drawer.open, .pal-scrim.open, #cfgElevate")) return;
  liveRefreshInFlight = true;
  try { await liveAdopt(true); } catch(e){ liveState.refreshFailed = true; }
  finally { liveRefreshInFlight = false; liveFreshness(); }
}
function liveStartRefresh(){
  if(liveRefreshTimer) return;
  liveRefreshTimer = setInterval(liveRefresh, LIVE_REFRESH_MS);
  setInterval(liveFreshness, 5000);                 // the AGE ticks even between refreshes
  document.addEventListener("visibilitychange", ()=>{ if(document.visibilityState==="visible") liveRefresh(); });
}

/* Workflows view gains a LIVE approvals strip: the POLL_PAUSE decisions the Runner is waiting on
   (REQ-519), each with Approve/Deny that POSTs the operator vote (REQ-518/INV-12). The vote binds the
   sealed action_id the poll presented — the Runner accepts it ONLY for that action. Mutation stays OFF:
   an approved action is recorded on the ledger and still refused at the mutation gate until the Phase-2
   flip, so the console can never itself cause an estate change. */
async function liveRefreshDecisions(){
  try{ const p = await liveGet("/v1/decisions"); liveState.decisions = p.decisions||[]; }
  catch(e){ liveState.decisions = null; }
}
async function liveVote(ref, actionID, approve, btn){
  const row = btn && btn.closest(".appr-row");
  const msg = row && row.querySelector(".appr-msg");
  if(row) row.querySelectorAll("button").forEach(b=>{ b.disabled = true; });
  const reenable = ()=>{ if(row) row.querySelectorAll("button").forEach(b=>{ b.disabled = false; }); };
  try{
    const r = await fetch("/api/v1/vote", { method:"POST", credentials:"same-origin",
      headers:{ "Content-Type":"application/json" },
      body: JSON.stringify({ external_ref:ref, action_id:actionID, approve:!!approve }) });
    if(r.status===202){
      if(msg) msg.textContent = approve ? "approved — recorded on the ledger (still refused at the actuation gate unless the mode permits)" : "denied — session stood down";
      await liveRefreshDecisions(); route("workflows");
    } else if(r.status===409){
      if(msg) msg.textContent = "this decision window is already closed";
      await liveRefreshDecisions();
    } else if(r.status===403){
      if(msg) msg.textContent = "cross-origin vote rejected"; reenable();
    } else if(r.status===429){
      if(msg) msg.textContent = "rate limited — retry in a moment"; reenable();
    } else {
      if(msg) msg.textContent = "vote delivery unavailable — retry"; reenable();
    }
  }catch(e){ if(msg) msg.textContent = "network error — retry"; reenable(); }
}

/* The decision-tracer inspector (spec/020 REQ-2011/2012): the console's face on GET /v1/sessions/{ref}.
   The audit-spine session list (/v1/sessions) drives a selectable index; picking one fetches the per-session
   detail endpoint and renders the REAL governed walk — classify -> propose -> predict -> verify — from the
   endpoint, in place of the WF_RUNS design fixtures. Observe-only: every field shown is a committed spine
   value (INV-15 — a step with no verdict reads "pending", never a fabricated one), no secret ever (INV-13). */
/* ===================== decision tracer, wired into the BEAUTIFUL wfView =====================
   Owner directive: keep the WF_RUNS 10-node design (modules/workflows/js.txt — wfView/wfNodeEl/wfPayEl/
   wfCmpEl/wfKidEls consumed UNCHANGED) and feed it REAL /v1/sessions data. dtoToWfRun() is the sole pure
   mapper: it projects one SessionDetailDTO (+ its list row) onto the exact WF_RUNS run/node contract, so
   nothing wfHead/wfNodeEl read is ever undefined. Observe-only: every field is a committed spine value; the
   five stages the spine does not yet persist (ingest/correlate/rag/screen/execute) render as MUTED, HONEST
   "no data yet" scaffolds — never a fabricated verdict/gate (INV-15). As the write-side MRs land, those
   scaffolds light up with real data and this frontend contract does not change. */

/* vocab remap: server enums -> the wfView render vocabulary */
const WF_BAND = { AUTO:"auto", AUTO_NOTICE:"notice", POLL_PAUSE:"pause" };
const WF_RISK_SCORE = { low:0.2, medium:0.5, high:0.8, critical:0.95 };
const WF_NODE_ST = { ok:"done", pending:"pending", deviation:"deviation", partial:"partial", refuse:"floored", deny:"floored", failure:"failed" };
function wfBandOf(b){ return WF_BAND[b] || "pause"; }
function wfNodeStOf(st){ return WF_NODE_ST[st] || (st ? "done" : "pending"); }
function wfKidStOf(st){ return (st==="failure"||st==="failed") ? "failed" : "done"; }
/* the payload lane colour is assigned by the WF STAGE, not by the spine kind */
function wfSrcOf(t){ return ({ ingest:"terr", correlate:"terr", rag:"terr", verify:"terr", execute:"haz" })[t] || "map"; }

/* the ten canonical stages, in timeline order, and the source each scaffold awaits */
const WF_STAGES = ["ingest","correlate","rag","classify","commit","agent","screen","gate","execute","verify"];
const WF_SCAFFOLD_SRC = { ingest:"the durable alert-log", correlate:"correlation evidence", rag:"retrieval evidence",
  screen:"the signal screen", execute:"the dry-run trail", classify:"the risk classifier", commit:"the prediction",
  agent:"the agent loop", gate:"the interceptor gate chain", verify:"the mechanical verdict" };
/* execute + verify are absent for a DIFFERENT reason than the other scaffolds: under mutation-OFF (Shadow)
   nothing actuates, so there is no execution to record and no post-execution verdict to compute. That is the
   GOVERNED SAFETY POSTURE, not a missing-data gap — label it as such so the walk reads "held by Shadow", never
   "broken". (verify only scaffolds when a session has no verdict; a completed one shows the real verdict.) */
const WF_SHADOW_STEPS = { execute:1, verify:1 };
function wfScaffold(t){
  // The honest note rides in the LABEL so a muted, un-instrumented stage reads at a glance (the collapsed
  // header shows it) — never a fabricated verdict/gate/pay (INV-15). The fuller note expands on click.
  if(WF_SHADOW_STEPS[t]){
    return { t, st:"pending", src:wfSrcOf(t), _synth:true, _shadow:true,
      lb: t==="execute" ? "execute · not run (Shadow)" : "verify · awaiting execution",
      pay: t==="execute"
        ? "Nothing actuates under Shadow — the mode withholds execution, so there is no execution to record. The real effect and its dry-run trail populate only when the mode is escalated owner-present."
        : "The mechanical verdict is computed only AFTER a governed action executes. Under Shadow mode nothing executed, so there is no verdict yet — this populates once an action runs and is verified." };
  }
  return { t, lb:t+" · no data yet", st:"pending", src:wfSrcOf(t), _synth:true,
    pay:(WF_SCAFFOLD_SRC[t]||"this stage")+" is not recorded to the spine yet — it lights up as the write-side lands" }; }

/* dtoToWfRun — the pure real->WF_RUNS projection. detail may be null (a list-only summary before the walk is
   fetched); every unconditional field still gets a safe, honest default so the beautiful header never throws. */
function dtoToWfRun(detail, row){
  const d = detail || {};
  const band = wfBandOf(d.band || (row && row.band));
  const nodes = Array.isArray(d.nodes) ? d.nodes : [];
  const first = k => nodes.find(n => n.t===k);

  const cl = first("classify");
  const classifyNode = cl
    ? { t:"classify", lb:cl.lb||"classify", src:"map", st:wfNodeStOf(cl.st), ts:cl.ts, band:wfBandOf(cl.band)||band, pay:cl.pay }
    : wfScaffold("classify");

  // commit = the sealed-action / prediction commit boundary. Sourced ONLY from a REAL committed artifact —
  // the assembler emits a predict step whenever a manifest was sealed (plan_ops) OR a consequence prediction
  // was committed. NEVER keyed on d.action alone (the classification's content hash exists for every classified
  // session — rendering a commit from it fabricates a prediction that was never committed, INV-15). The action
  // hash is shown only when a real commit boundary exists.
  const pr = first("predict");
  const prSealed = !!(pr && pr.lb==="Sealed action");   // a manifest was sealed but no consequence prediction committed
  const commitNode = pr
    ? { t:"commit", lb: prSealed ? "sealed action" : "prediction.commit", src:"map",
        // a sealed end-state is COMMITTED (done); only an unscored prediction awaiting the verify window is pending.
        st: prSealed ? "done" : wfNodeStOf(pr.st), ts:pr.ts, hash:d.action||"", pay: pr.pay,
        // the sealed action's structured end-state ops [{op,t}] — wfPayEl renders these as add/change/destroy
        plan: (Array.isArray(pr.plan_ops) && pr.plan_ops.length) ? pr.plan_ops : undefined }
    : wfScaffold("commit");

  // agent = fold the flat agent-cycle (+credential) steps into ONE node with nested tool-call kids.
  // Prefer a real detail.agent_kids[] when a future backend MR ships one (the fold is the fallback).
  const propose = first("propose");
  const cycles = nodes.filter(n => n.t==="agent-cycle" || n.t==="credential");
  let agentNode;
  if(Array.isArray(d.agent_kids) && d.agent_kids.length){
    agentNode = { t:"agent", lb:"agent-loop", src:"map", st:"done", ts:cycles[0]&&cycles[0].ts,
      pay: propose ? propose.pay : (d.agent_kids.length+" tool call(s)"), kids:d.agent_kids };
  } else if(cycles.length || propose){
    agentNode = { t:"agent", lb:"agent-loop", src:"map", st:"done", ts:(cycles[0]&&cycles[0].ts)||(propose&&propose.ts),
      pay: propose ? propose.pay : (cycles.length+" reasoning cycle(s)"),
      kids: cycles.map(c => ({ tool:(c.plan&&c.plan[0]) || (c.t==="credential"?"credential":"agent"),
        q:c.lb||"", res:c.pay||"", dur:null, retry:0, st:wfKidStOf(c.st) })) };
  } else agentNode = wfScaffold("agent");

  // gate = the WHOLE ordered AUTHORIZATION gate chain (interceptor_gate_verdict) + policy decision. Fold every
  // gate row, not just the first, so a LATE floor (e.g. the mode chokepoint) is never masked by an earlier
  // admission pass. The interceptor's trailing 'verify' gate carries the MECHANICAL VERDICT (match/partial/
  // deviation), not an authorization outcome — it is rendered at the verify slot and EXCLUDED here so a clean
  // deviation is never miscast as a governance floor. A gate floors the chain only on an explicit refusal.
  const gateNodes = nodes.filter(n => n.t==="gate" && (n.gate||"").toLowerCase() !== "verify" && !/verify/i.test(n.lb||""));
  const policy = first("policy");
  let gateNode;
  if(gateNodes.length || policy){
    const floored = gateNodes.filter(g => /refuse|deny|block/i.test(g.verdict||""));
    const anyFloored = floored.length > 0 || (policy && policy.verdict==="deny");
    // passed = no floor AND either an authorization gate resolved or the policy allowed; an unknown policy
    // verdict with no gates is NEITHER passed nor floored (neutral — never a fabricated floor, MINOR-3).
    const decided = gateNodes.length > 0 || (policy && (policy.verdict==="auto"||policy.verdict==="approve"));
    const passed = !anyFloored && decided;
    const g0 = gateNodes[0] || policy;
    const chainPay = gateNodes.length
      ? gateNodes.map(g => "· "+(g.gate||g.lb||"gate")+": "+(g.verdict||"?")).join("\n")
      : (policy && policy.pay);
    gateNode = { t:"gate", lb: gateNodes.length>1 ? (gateNodes.length+"-gate chain") : ((g0&&g0.lb)||"gate"),
      src:"map", st: anyFloored?"floored":wfNodeStOf(g0&&g0.st), ts:(g0&&g0.ts),
      // pass | floored | (neutral, when a decision boundary exists but neither passed nor refused)
      gate: anyFloored ? "floored" : (passed ? "pass" : undefined), pay:chainPay, hash:(g0&&g0.hash) };
  } else gateNode = wfScaffold("gate");

  // verify = the mechanical verdict
  //
  // ★ THE ROW IS THE THIRD SOURCE, AND IT WAS MISSING. A LIST row is built as dtoToWfRun(null, row), so
  // `detail` is null, `d` is {}, and `nodes` is empty — meaning both `vf` and `d.verdict` are always
  // undefined for every row in the governed-run list. The verdict resolved to null on all of them, and a
  // run the deterministic verifier marked DEVIATION rendered exactly like a clean one. /v1/sessions has
  // carried `verdict` on the row the whole time (core/httpapi/sessions.go:39, "match | partial |
  // deviation, or empty when no verdict exists yet"); nothing read it.
  //
  // Order is deliberate: the walked node wins, then the detail, then the row. The first two are the
  // fetched evidence for THIS run; the row is the spine's own summary and is the only one a list-only
  // render has. Empty stays empty — a session with no verdict must still render "pending", never an
  // invented outcome (INV-15).
  const vf = first("verify");
  const verifyVerdict = (vf && vf.verdict) || d.verdict || (row && row.verdict) || "";
  const verifyNode = vf
    ? { t:"verify", lb:vf.lb||"verify", src:"terr", st:wfNodeStOf(vf.st), ts:vf.ts, verdict:verifyVerdict, pay:vf.pay }
    : wfScaffold("verify");

  // screen = the admission-time signal screen (session_risk_audit.signals_json): the never-auto floor / blast-radius
  // / poll_reason signals that drove the band. Real when the classifier recorded any; else the honest scaffold.
  const sc = first("screen");
  const screenNode = sc
    ? { t:"screen", lb:sc.lb||"admission screen", src:"map", st:wfNodeStOf(sc.st), ts:sc.ts, pay:sc.pay }
    : wfScaffold("screen");

  // rag = the retrieval context (session_triage.skill_loads): the composed-seed provenance the proposal was
  // built from. The skill/precedent loads ride in node.plan (the DTO's plan lane); render them as the pay so
  // the muted stage lights up with real retrieval evidence. Real when loads exist; else the honest scaffold.
  const rg = first("rag");
  const ragLoads = rg && Array.isArray(rg.plan) ? rg.plan : [];
  const ragNode = (rg && ragLoads.length)
    ? { t:"rag", lb:rg.lb||"rag-context", src:"terr", st:wfNodeStOf(rg.st), ts:rg.ts,
        pay:(rg.pay?rg.pay+"\n":"")+"composed seed:\n· "+ragLoads.join("\n· ") }
    : wfScaffold("rag");

  // execute = the actual actuation. Lit from the execute-named interceptor gate verdict when a governed
  // actuation traversed it (mutation ON): pass ⇒ the effect ran. Under Shadow (mutation OFF) nothing actuates,
  // so there is no execute gate row and the honest "not run (Shadow)" scaffold shows — never a false Shadow
  // label over a real actuation.
  const execGate = nodes.find(n => n.t==="gate" && /execute/i.test(n.gate||n.lb||""));
  const executeNode = execGate
    ? { t:"execute", lb: execGate.verdict==="pass" ? "execute · ran" : ("execute · "+(execGate.verdict||"?")),
        src:"haz", st:wfNodeStOf(execGate.st), ts:execGate.ts,
        pay: execGate.verdict==="pass" ? "action executed — execute gate passed" : ("execute gate: "+(execGate.verdict||"?")) }
    : wfScaffold("execute");

  // ingest = the durable front-door alert (ingest_alert): the source · rule · severity · host + summary the
  // ingest tier admitted. Real when a durable record exists; else a light honest scaffold carrying only the
  // front-door identity we DO have (a pull-minted session bypassed the front door — no durable record).
  const ig = first("ingest");
  const ingestNode = ig
    ? { t:"ingest", lb: ig.lb||"ingested", src:"terr", st:wfNodeStOf(ig.st), ts:ig.ts, pay: ig.pay }
    : { t:"ingest", lb:"ingested", src:"terr", st:"done", _synth:true,
        pay:"alert "+(d.ref||(row&&row.external_ref)||"")+(d.host?(" · "+d.host):"")+"\n(front-door detail not persisted — pull-minted or pre-durable-log session)" };

  // correlate = the deterministic dedup/suppression decision (governance_ledger "suppress:"+ref). Real when
  // the run recorded one; else the honest scaffold (a session predating the suppression-ledger wiring).
  const co = first("correlate");
  const correlateNode = co
    ? { t:"correlate", lb: co.lb||"correlate", src:"terr", st:wfNodeStOf(co.st), ts:co.ts,
        pay:(co.verdict==="clean"?"escalated — proceeded to investigation (not suppressed)\n":"")+(co.pay||"") }
    : wfScaffold("correlate");

  const slots = { ingest:ingestNode, correlate:correlateNode, rag:ragNode,
    classify:classifyNode, commit:commitNode, agent:agentNode, screen:screenNode,
    gate:gateNode, execute:executeNode, verify:verifyNode };

  const verdict = (verifyVerdict==="match"||verifyVerdict==="partial"||verifyVerdict==="deviation") ? verifyVerdict : null;
  const riskLabel = String(d.risk || (row && row.risk_level) || "").toLowerCase();
  const terminal = d.status==="executed" || d.status==="stopped";
  let status = ({ received:"running", classified:"running", proposed:"running", executed:"done", stopped:"failed" })[d.status] || "done";
  if(band==="pause" && !verdict) status = "waiting";      // POLL_PAUSE awaiting a human vote
  if(gateNode.gate==="floored") status = "floored";

  return {
    id: d.id || d.ref || (row && row.external_ref) || "",
    ref: d.ref || d.id || (row && row.external_ref) || "",
    title: d.title || d.ref || (row && row.external_ref) || "",
    host: d.host || "", band, status,
    /* spec/029 T-029-5 (REQ-2906): the session's armed-revert window, straight off the walk DTO's
       commit_confirm enrichment. null when the DTO carries none (the ordinary non-eligible run) —
       the header renders no chip, never a fabricated one. */
    commit_confirm: d.commit_confirm || null,
    live: !terminal && !!detail,          // only a loaded, non-terminal walk streams
    /* ★ FIVE HEADER VALUES THE SESSION DTO DOES NOT CARRY, EACH DEFAULTED TO A CONFIDENT NUMBER. This is
       the header an operator reads BEFORE APPROVING AN IRREVERSIBLE ACTION, and every default fell on the
       reassuring side: `reversible` defaulted to TRUE, so the dangerous state could never render at all;
       an unknown risk label became a 0.40 meter that looks measured; absent conf and blast became 0,
       indistinguishable from a real measured zero. NULL is now the fourth state and the renderer prints
       "—" for it. `unknown` and `safe` are not the same claim, and only one of them is ever honest here.

       elapsed: derived from the run's REAL classified_at rather than counted up client-side. It used to be
       a hardcoded 0 that the ticker never touched on live runs, so "how long has this been stuck awaiting
       approval" always answered "0s" — which reads as "it just started". */
    startedMs: (row && row.classified_at) ? (Date.parse(row.classified_at) || null) : null,
    elapsed: null,
    risk: (riskLabel in WF_RISK_SCORE) ? WF_RISK_SCORE[riskLabel] : null,
    conf: (d.conf === undefined || d.conf === null || d.conf === "" || isNaN(Number(d.conf))) ? null : Number(d.conf),
    exec_class: d.exec_class || "—",
    reversible: d.reversible !== undefined ? d.reversible : null,
    blast: (d.blast === undefined || d.blast === null) ? null : d.blast,
    action: d.action || "",
    verdict,
    nodes: detail ? WF_STAGES.map(s => slots[s]) : WF_STAGES.map(wfScaffold),
    _loaded: !!detail,
  };
}

/* liveWfRuns builds the run list from the adopted /v1/sessions rows, with STABLE OBJECT REFERENCES so a
   fetched walk merges in place across re-renders. It is the array wfView renders when live.

   ★ IT USED TO CACHE FOREVER. `if(liveState._wfRuns) return liveState._wfRuns;` was set on the first
   successful /v1/sessions read and never invalidated, so the governed-run list — and the decision tracer
   built from it — was frozen at whatever the estate looked like at boot. Every 25s poll refreshed
   liveState.sessions underneath it and changed nothing on screen: new governed runs never appeared, and runs
   that had since completed or aged out kept being offered as selectable. On the surface whose entire purpose
   is auditing what the system DID, that is the worst place for a stale list.
   The cache was there for a real reason (identity must survive a re-render or the open walk collapses), so
   the fix keeps identity and drops the staleness: re-derive on every call, but REUSE the existing object for
   any run still present, keyed by external_ref. Rows that persist keep their reference and their merged walk;
   rows that are gone disappear; new rows appear. */
function liveWfRuns(){
  const rows = liveState.sessions || [];
  const prev = liveState._wfRuns || [];
  // A one-pass index of what we already have, so identity is preserved without an O(n^2) scan.
  const byRef = new Map();
  prev.forEach(r => { const k = r && r._row && r._row.external_ref; if(k) byRef.set(k, r); });
  liveState._wfRuns = rows.map(row => {
    const hit = row && row.external_ref ? byRef.get(row.external_ref) : null;
    if(hit){ hit._row = row; return hit; }   // same run: keep the object (and any walk merged into it)
    const r = dtoToWfRun(null, row); r._row = row; return r;
  });
  return liveState._wfRuns;
}

/* wfOnSelect — the wfView select hook (workflows/js.txt wfSelect): lazy-load the chosen run's walk from the
   detail endpoint, map it in place, re-render, and stream if the run is still live. */
async function wfOnSelect(ref){
  const run = liveWfRuns().find(r => r.id===ref);
  // `_failed` LATCHES A FAILED LOAD, and without it this hook is an unbounded retry loop that runs at
  // NETWORK SPEED. Measured live 2026-07-28: a single idle tab left on a run whose detail endpoint answers
  // 403 produced 28,094 rejected requests against one success — ~75 req/s, sustained for hours, through the
  // same nginx that carries the LibreNMS and Alertmanager ingest lanes. The mechanism is a cycle, not a
  // retry policy: the catch below clears `_loading` and calls route(), route() re-renders the workflows
  // view, the view calls its wfSelect hook for the selected run, and the guard above lets it straight
  // through again because `_loaded` is still false and `_loading` is false once more. Nothing in the loop
  // waits, so it spins as fast as the network answers.
  //
  // The latch is per page-load and deliberately covers EVERY error, not only auth: a transient 5xx that
  // re-arms itself on the next render is the same storm with a different status code. The rendered error
  // state tells the operator to retry, and a reload (or selecting another run and returning) is the retry —
  // an explicit human act, not an implicit hot loop.
  if(!run || run._loaded || run._loading || run._failed) return;
  run._loading = true;
  let dto;
  try{ dto = await liveGet("/v1/sessions/"+encodeURIComponent(ref)); }
  catch(e){ run._loading = false; run._failed = true; run._err = (e&&e.status)||"err"; route("workflows"); return; }
  Object.assign(run, dtoToWfRun(dto, run._row), { _loading:false, _failed:false, _err:undefined, _row:run._row });
  route("workflows");
  if(run.live && window.EventSource) liveOpenWfStream(ref);
}
/* liveOpenWfStream subscribes to GET /api/v1/sessions/{ref}/stream (same-origin cookie auth, the elevated
   trace-read role). The server re-assembles the walk from the durable spine and pushes the FULL current walk
   as a `snapshot` event whenever it changed — a new boundary committed OR a step enriched in place — so the
   client re-maps it into the run and re-renders. Real boundary rows only, never a client-side clock; it closes
   on the server's `done` event. */
let liveWfES = null;
function liveCloseWfStream(){ if(liveWfES){ try{ liveWfES.close(); }catch(e){} liveWfES = null; } }
function liveOpenWfStream(ref){
  liveCloseWfStream();
  let es; try{ es = new EventSource("/api/v1/sessions/"+encodeURIComponent(ref)+"/stream"); }catch(e){ return; }
  liveWfES = es;
  es.addEventListener("snapshot", ev=>{
    let dto; try{ dto = JSON.parse(ev.data); }catch(e){ return; }
    const run = liveWfRuns().find(r => r.id===ref); if(!run) return;
    Object.assign(run, dtoToWfRun(dto, run._row), { _loaded:true, _err:undefined, _row:run._row });
    if(wfSelId===ref) route("workflows");
  });
  es.addEventListener("done", ev=>{
    let d; try{ d = JSON.parse(ev.data); }catch(e){ d = {}; }
    const run = liveWfRuns().find(r => r.id===ref);
    if(run && d.status){ run.status = d.status==="executed"?"done":d.status==="stopped"?"failed":run.status; run.live = false; }
    liveCloseWfStream();
    if(wfSelId===ref) route("workflows");
  });
  es.onerror = ()=>{ liveCloseWfStream(); }; /* server closed / session gone — stop, no client clock */
}

async function cfgWriteFail(r){
  let server=""; try{ server=(await r.text()).trim().slice(0,200); }catch(e){}
  if(r.status===401) return { elevate:true, msg:"admin elevation required — step up to write" };
  if(r.status===403) return { msg:"cross-origin write rejected" };
  if(r.status===404&&/unknown configuration key/.test(server)) return { msg:"refused: "+server };
  if(r.status===404||r.status===405||r.status===503) return { msg:"write path not deployed on this control plane yet" };
  if(r.status===422) return { msg:"refused: "+(server||"pinned by law — the clamp is the law") };
  if(r.status===429) return { msg:"rate limited — retry in a moment" };
  if(r.status===400) return { msg:"refused: "+(server||"a rationale is required") };
  return { msg:"write failed ("+r.status+")"+(server?" — "+server:"") };
}

async function cfgSaveKnob(name, value, rationale, msgEl){
  msgEl.style.color="var(--ink3)"; msgEl.textContent="recording…";
  let r=null;
  try{
    r = await fetch("/api/v1/config/"+encodeURIComponent(name),{ method:"POST", credentials:"same-origin",
      headers:{ "Content-Type":"application/json" }, body: JSON.stringify({ value:value, rationale:rationale }) });
  }catch(e){}
  if(r && r.ok){
    const out = await r.json().catch(()=>({}));
    msgEl.style.color="var(--match)";
    msgEl.textContent="recorded — ledger seq "+(out.ledger_seq!=null?out.ledger_seq:"?")+" · source=console";
    cfgUI.edit=null;
    await liveRefreshConfigSecrets();
    return;
  }
  const f = r ? await cfgWriteFail(r) : { msg:"network error — retry" };
  msgEl.style.color="var(--deviation)"; msgEl.textContent=f.msg;
  if(f.elevate) cfgElevateOverlay("That write needs an elevated session.");
}

/* The "API" view (REQ-515): the control plane's own endpoint map, rendered natively.
   Auth class is derived from the path — machine-only (ingest/replay) vs session-ok reads. */
/* ★ THE ACCESS COLUMN ASSERTED "GET" ON ROUTES THAT HAVE NO GET. The fallback label was the literal
   "session-ok · GET", so seven POST-only routes — including /v1/mode, the actuation-mode chokepoint —
   were published in the API surface as session-authenticated READS. This is the same defect the Methods
   column carried and had fixed: a constant printed where a per-row fact belongs, just moved one column
   right. The methods are already parsed from the OpenAPI document one function away; the label is derived
   from them now, so a route's access class cannot disagree with the verbs it actually exposes. */
function apiAuthClass(path, methods){
  if(/\/v1\/ingest\//.test(path) || /\/replay$/.test(path)) return ["machine-only","var(--notice)"];
  if(/\/v1\/session(\/logout)?$/.test(path)) return ["operator-login","var(--map)"];
  const ms = (methods||[]).filter(m=>m!=="connect"&&m!=="head"&&m!=="options"&&m!=="trace");
  if(!ms.length) return ["session-ok","var(--match)"];
  const writes = ms.filter(m=>m!=="get");
  // A write route is not the same risk as a read route and must not be captioned as one.
  if(!writes.length) return ["session-ok · GET","var(--match)"];
  if(writes.length === ms.length) return ["session-ok · "+writes.map(m=>m.toUpperCase()).join("/"),"var(--notice)"];
  return ["session-ok · GET + "+writes.map(m=>m.toUpperCase()).join("/"),"var(--notice)"];
}
/* Parse the OpenAPI `paths:` block from the raw YAML into {path:[methods]}. Deliberately minimal:
   two indent levels under `paths:` — the path key, then the method keys. */
function apiParsePaths(yaml){
  const out={}; const lines=(yaml||"").split("\n");
  let inPaths=false, cur=null;
  for(const ln of lines){
    if(/^paths:\s*$/.test(ln)){ inPaths=true; continue; }
    if(inPaths && /^\S/.test(ln)){ break; } // dedent out of paths:
    if(!inPaths) continue;
    let m=ln.match(/^  (\/\S+):\s*$/);
    if(m){ cur=m[1]; out[cur]=out[cur]||[]; continue; }
    m=ln.match(/^    (get|post|put|delete|patch|head|options|connect|trace):\s*$/);
    if(m && cur){ if(out[cur].indexOf(m[1])<0) out[cur].push(m[1]); }
  }
  return out;
}

/* Knowledge (the living wiki, REQ-derived): on a live session the pages are COMPOSED from the real
   estate graph, the incidents on the audit spine, and their verdicts — never the fixture. Each host
   page joins the estate edges; each incident precedent is a recorded session; the operational rules
   are the system's standing invariants. */
function knowLiveModel(){
  const est = liveState.estate || {};
  const nodes = (est.available && Array.isArray(est.nodes)) ? est.nodes : [];
  const edges = (est.available && Array.isArray(est.edges)) ? est.edges : [];
  const sess = Array.isArray(liveState.sessions) ? liveState.sessions : [];
  /* ★ THIS READ A FIELD THAT NEVER EXISTED, SO EVERY HOST PAGE SHOWED ZERO INCIDENTS. The DTO carried no
     host, and `signals` — the only place this looked — holds novelty_key / poll_reason / novelty_count /
     actor_attribution. Measured on production: 0 of 50 recent sessions carry a `host` key, so `hostOf`
     returned "" for every row and the per-host filter below matched NOTHING, for every host, always —
     while session_triage.host was populated on 49 of those same 50 rows and the spine held 3,202 sessions
     across 78 hosts. The page still captioned itself as composed from the incidents that touched the
     entity. `s.host` is now the real field; the signals fallbacks stay for pre-fix rows. */
  const hostOf = s => s.host || (s.signals && (s.signals.host||s.signals.target||s.signals.device||s.signals.hostname)) || "";
  const incNote = s => {
    const v=s.verdict, b=s.band;
    if(v==="deviation") return "outcome deviated from the committed prediction — clamped never-auto";
    if(v==="match") return "committed prediction matched the observed outcome";
    if(v==="partial") return "partial match; remediation held human-gated until a clean match";
    return b==="POLL_PAUSE" ? "held at the never-auto floor for a human" : "classified read-only";
  };
  const hosts = nodes.map(n=>({
    name:n.name, type:n.type, role:n.type,
    depends_on: edges.filter(e=>e.from===n.name).map(e=>({to:e.to,rel:e.rel,confidence:e.confidence,source:e.source})),
    dependents: edges.filter(e=>e.to===n.name).map(e=>({from:e.from,rel:e.rel,confidence:e.confidence,source:e.source})),
    incidents: sess.filter(s=>hostOf(s)===n.name).map(s=>({ref:s.external_ref,band:s.band,verdict:s.verdict,note:incNote(s)})),
  }));
  const incidents = sess.map(s=>({
    ref:s.external_ref, band:s.band, risk:s.risk_level, verdict:s.verdict, host:hostOf(s),
    classified: s.classified_at, learned: incNote(s) + (s.action_id?(" · action "+String(s.action_id).slice(0,8)):""),
  }));
  /* Rules came from the KNOW FIXTURE unconditionally, so with no live estate and no live sessions this
     model still reported a non-zero population and the rail badge published it (3, measured live under
     full API starvation). The knowledge model must be built from live reads only; the view's own
     un-adopted path already renders the fixture behind a "design preview" label, which is where fixture
     content belongs. */
  /* ★★★ AND THEY ARE STILL HAND-AUTHORED, ON A VIEW WHOSE HEADER SAYS NOTHING IS.
     The guard above stopped the fixture rules from surviving API starvation, which was the right fix for
     THAT defect and left this one: KNOW.rules is three literal sentences in modules/knowledge/fixtures.txt,

  const rules = (nodes.length || sess.length) && (typeof KNOW!=="undefined" && KNOW.rules)

  return { hosts, incidents, rules };
}

/* Live estate graph: a MEANINGFUL hub-and-spoke topology built from the worker's published
   /v1/estate edge list. There is NO nodes array — nodes are derived from the unique from/to of the
   edges. Same visual language as the fixture estateGraph(): percent-coordinate SVG (so strokes stay
   uniform), dashed site-region rectangles ("dc1" / "dc2"), absolutely-positioned HTML dots +
   labels, edges as lines weighted by confidence. Hubs = high in-degree (the Proxmox hosts that
   guests run_on); each hub's dependents fan around it. Layout is fully deterministic (index-based
   angular placement — no Math.random / Date.now — so it never jitters on re-render). */
function liveEstateGraph(e){
  const NS="http://www.w3.org/2000/svg";
  const g=h("div",{class:"graph",style:"height:540px;position:relative;border:1px solid var(--line2);border-radius:var(--radius);overflow:hidden;margin-bottom:16px"});
  const svg=document.createElementNS(NS,"svg");
  svg.setAttribute("width","100%"); svg.setAttribute("height","100%");
  g.append(svg);

  const edges=(Array.isArray(e.edges)?e.edges:[]).filter(ed=>ed&&ed.from&&ed.to);
  // ---- derive nodes + in/out-degree from the edge list ----
  const N=new Map();
  const nodeOf=nm=>{ let n=N.get(nm); if(!n){ n={name:nm,inDeg:0,outDeg:0,out:[]}; N.set(nm,n); } return n; };
  edges.forEach(ed=>{
    const conf=(typeof ed.confidence==="number")?Math.max(0,Math.min(1,ed.confidence)):0.7;
    const a=nodeOf(ed.from), b=nodeOf(ed.to);
    a.outDeg++; b.inDeg++; a.out.push({to:ed.to,conf:conf});
  });
  const all=[...N.values()];
  if(!all.length){
    g.append(h("div",{class:"empty",style:"position:absolute;inset:0;display:flex;flex-direction:column;align-items:center;justify-content:center"},
      h("div",{class:"big"},"Empty graph"),h("div",{class:"desc"},"The published snapshot carries no edges.")));
    return g;
  }
  const siteOf=nm=>{ const p=String(nm).slice(0,5); return p==="nllei"?"dc1":p==="grskg"?"dc2":"other"; };

  // ---- hubs = high in-degree (the pve hosts guests run_on) ----
  const HUB_MIN=2, MAX_HUBS=16;
  let hubs=all.filter(n=>n.inDeg>=HUB_MIN).sort((a,b)=>b.inDeg-a.inDeg);
  if(!hubs.length){ // degenerate: no clear hub — promote the highest-in-degree nodes so it still reads
    hubs=all.filter(n=>n.inDeg>0).sort((a,b)=>b.inDeg-a.inDeg).slice(0,6);
  }
  if(hubs.length>MAX_HUBS) hubs=hubs.slice(0,MAX_HUBS);
  const hubSet=new Set(hubs.map(n=>n.name));

  // ---- assign each non-hub node to the hub it runs_on (highest-confidence incident edge) ----
  const depOf=new Map(); hubs.forEach(hb=>depOf.set(hb.name,[]));
  const placed=new Set(hubSet);
  all.forEach(n=>{
    if(hubSet.has(n.name)) return;
    let target=null,best=-1;
    n.out.forEach(o=>{ if(hubSet.has(o.to)&&o.conf>best){ best=o.conf; target=o.to; } });
    if(target){ depOf.get(target).push({node:n,conf:best}); placed.add(n.name); }
  });
  depOf.forEach(list=>list.sort((x,y)=>x.node.name<y.node.name?-1:x.node.name>y.node.name?1:0));
  const loose=all.filter(n=>!placed.has(n.name)).sort((x,y)=>x.name<y.name?-1:x.name>y.name?1:0);

  // ---- regions: one dashed rectangle per present site (unknown prefixes → a neutral zone) ----

/* liveLogLines is the terminal's live feed, rebuilt on each render from state the console already fetched
   (no new reads). ASCENDING time — the terminal's tail is its "newest", and the body sticks to it. An
   undated row sorts FIRST (away from the tail): it must never occupy the newest position. */
let liveLogWindow = null;
function liveBuildLogLines(){
  if(!liveState.on) { liveLogWindow = null; return null; }
  const evMs = s => { const t = Date.parse(s); return isNaN(t) ? null : t; };
  const lines = [];
  (liveState.ledger||[]).forEach(e=>{
    const kind = e.kind||"";
    lines.push({
      t: e.ts ? evMs(String(e.ts).replace(" ","T")+"Z") : null,
      actor: (e.actor==="human"||e.actor==="ai") ? e.actor : "system",
      /* An honest MAPPING, not a fabricated level: the ledger carries no severity, but a withheld entry or
         a dark-seam/config-gap report is the platform flagging its own blind spot — the one class of
         governance line an operator must not scroll past. Everything else is info. */
      lvl: (/dark|gap|withheld/.test(kind)) ? "warn" : "info",
      src: "governance",
      cid: "seq "+(e.seq!=null?e.seq:"?"),
      msg: kind + (e.ref ? " — "+e.ref : ""),
      session: null,
      detailTitle: kind+" · seq "+(e.seq!=null?e.seq:"?"),
      detail: e.ref || "(this ledger entry carries no detail text)",
    });
  });
  (liveState.alerts||[]).forEach(a=>{
    const sev = String(a.severity||"").toLowerCase();
    lines.push({
      t: evMs(a.received_at||a.observed_at||""),
      actor: "system",
      lvl: sev==="critical"||sev==="crit" ? "error" : (sev==="warning"||sev==="warn" ? "warn" : "info"),
      src: a.source_type||a.src||"ingest",
      cid: a.host||"",
      msg: "alert "+(a.alert_rule||"")+(a.host?" · "+a.host:"")+(a.external_ref?"  ["+a.external_ref+"]":""),
      session: a.external_ref||null,
    });
  });
  lines.sort((x,y)=>{
    if(x.t==null && y.t==null) return 0;
    if(x.t==null) return -1;   // undated first = furthest from the tail's "newest"
    if(y.t==null) return 1;
    return x.t - y.t;
  });
  liveLogWindow = lines.length ? lines : null;
  return liveLogWindow;
}

/* REBOUND: absolute UTC for live lines ("MM-DD HH:MM:SS" — the window spans days, so the date is not
   optional), the fixture's relative clock otherwise. The header states the zone once for the column. */
function logFmtTs(t){
  if(liveLogWindow){
    if(t==null) return "— undated —";
    return new Date(t).toISOString().slice(5,19).replace("T"," ");
  }
  const d = new Date(LOG_T0 + t), p = (n,w)=>String(n).padStart(w,"0");
  return p(d.getHours(),2)+":"+p(d.getMinutes(),2)+":"+p(d.getSeconds(),2)+"."+p(d.getMilliseconds(),3);
}

/* liveAge renders a sealed_at as the compact relative age the ribbon shows ("4m"), lowercase m for minutes
   — an uppercase M reads as months, which the fixture comment already warned about. */
function liveAge(iso){
  if(!iso) return "";
  const t = new Date(iso); if(isNaN(t.getTime())) return "";
  const s = Math.max(0, (Date.now()-t.getTime())/1000);
  if(s < 90) return Math.round(s)+"s";
  if(s < 5400) return Math.round(s/60)+"m";
  if(s < 172800) return Math.round(s/3600)+"h";
  return Math.round(s/86400)+"d";
}

   genuine per-step confidence. Nothing new was needed on the backend — the data was simply never read.

   MAPPING IS DELIBERATELY PARTIAL. The fixture's chain has six glyphs (alert / hypothesis / evidence /
   diagnosis / proposal / verdict); the spine records four of those honestly, so only those are emitted.

/* ===================== SELECTING A RECORDED WALK (TG-269) =====================
   The spine holds ~1,900 recorded sessions and this page could reach exactly ONE of them: whatever was
   newest at adopt. The view body contained no control of any kind — no selector, no handler, no fetch — so
   the pane never changed for the life of the tab. That is indistinguishable from static content to the
   operator looking at it, and that is precisely how it was reported.

   THE SELECTION LIVES IN location.search, NOT THE HASH. route() writes `location.hash = name` on every
   navigation and the hashchange listener does `if(views[n]) route(n)` — so a hash of "reasoning/<ref>"
   would be clobbered by the first and ignored by the second. The query string is orthogonal to both, which
   means deep-linking costs zero changes to the router. */
function liveReasonUrlRef(){
  try{ return new URLSearchParams(location.search).get("session") || null; }catch(e){ return null; }
}
/* Assembled from parts rather than through the URL constructor. Historical note: the self-containment
   guard (deploy/served_console_test.go) once scanned the WHOLE document for CSS url() references, so the
   constructor — and even prose describing it — read as an external stylesheet; it cost two red pipelines
   on 2026-08-03 (TG-270). The guard is now scoped to CSS contexts and pinned by tests, so either spelling
   is safe. This one stays because it is independently better: keeping pathname and hash explicit
   documents that only the query is being touched. */
function liveReasonSetUrlRef(ref){
  try{
    const q = new URLSearchParams(location.search);
    if(ref) q.set("session", ref); else q.delete("session");
    const s = q.toString();
    history.replaceState(null, "", location.pathname + (s ? "?"+s : "") + location.hash);
  }catch(e){}   // a browser that refuses the history write still gets a working picker, just not a citable URL
}
/* One read, one place. 403 is recorded as the sentinel "forbidden" (a real answer from the trace gate), any
   other failure as null (we do not know) — the view renders those two differently and must keep being able
   to tell them apart. */
async function liveReasonLoad(ref){
  try{ liveState.reasonDto = await liveGet("/v1/sessions/"+encodeURIComponent(ref)); }
  catch(e){ liveState.reasonDto = (e && e.status === 403) ? "forbidden" : null; }
  /* THE CLAIM TRAVELS WITH THE WALK IT BELONGS TO (TG-201). Loading it here — in the ONE place that loads a

// host leads because that is what an operator types to search a native select; ref trails as the identifier.
function liveReasonLabel(r){
  const ts = (r.classified_at || "").replace("T"," ").slice(0,16);
  return [r.host || "(no host)", ts, r.band || "unbanded", r.external_ref].filter(Boolean).join("  ·  ");
}
/* ★ THESE FIELD NAMES ARE THE SERVER'S, NOT A GUESS. Every one of them is read off SessionNodeDTO /
   SessionDetailDTO in core/httpapi/session_detail.go. The first version of this function read dto.steps[]
   with st.kind / st.reason / st.label / st.confidence / st.tools[], and the view guarded on
   dto.external_ref — SEVEN field names, of which the server sends NONE. The DTO is {id, ref, title, host,
   band, status, risk, conf, action, verdict, nodes[]} and a node is {t, lb, ts, st, src, pay, plan[],
   gate, band, hash, verdict, conf, min_conf}.

   The consequence was total and silent: `if(!dto.external_ref) return liveOrigReasoning()` was ALWAYS
   true, so this whole surface fell back to the labelled fixture on every load since it shipped, while
   liveState.reasonDto sat fully populated beside it. Verified live 2026-07-29 on
   territory-grounder.example.net — the #reasoning view rendered "FIXTURE · REPRESENTATIVE, NOT
   LIVE" over an invented host, with six real recorded cycles available in memory.

   Keep them aligned with the Go struct tags, not with what a field "should" be called. */
function liveReasonChain(dto){
  const out = [];
  (dto.nodes||[]).forEach(n=>{
    const kind = REASON_KIND[n.t];
    if(!kind) return;                       // a boundary with no honest glyph is omitted, never recoloured
    /* A CYCLE THAT RECORDED NO THOUGHT IS NOT A HYPOTHESIS (TG-269). The guard below used to read
       `n.pay || n.lb` and the comment claimed "an empty step is not a reasoning step" — but lb is
       BOILERPLATE ("ReAct cycle 6"), never empty, so the guard could not fire and the boilerplate rendered
       as the agent's thought. Live on 2026-08-03 the newest walk showed a hypothesis reading, in full,
       "ReAct cycle 6". Dropping the node instead would hide a boundary that really happened, so it is
       emitted as an explicit GAP: reasoningZone falls back to the neutral "··" glyph for a kind it does not
       know, which is already the design's answer for an unclassifiable boundary. */
    if(n.t === "agent-cycle" && !n.pay){
      out.push({ kind:"unrecorded", conf:null, cite:"",
        t:"This cycle ran but recorded no thought"+(n.lb ? " ("+n.lb+")" : "")+" — the boundary is real, its reasoning was not captured." });
      return;
    }
    const text = n.pay || n.lb || "";
    if(!text) return;                       // an empty step is not a reasoning step
    out.push({
      kind, t: text,
      // conf is the STATED confidence at that boundary; 0 means "not stated" in this DTO, so it is dropped
      // rather than drawn as total no-confidence.
      conf: (typeof n.conf === "number" && n.conf > 0) ? n.conf : null,
      // plan[] carries the step's provenance — for an agent cycle that is the tool it reached for.
      cite: (n.plan && n.plan.length) ? n.plan[0] : "",
      // ev is the handle for the stored GROUND TRUTH behind this step (TG-272). Empty for every session
      // recorded before evidence capture existed, which the citation states rather than hides.
      ev: n.ev || "",
    });
  });
  return out;
}

async function liveOpenEvidence(ref, ev, tool){
  const d = $("#drawer"); if(!d) return;
  const paint = body => {
    d.innerHTML=""; d.removeAttribute("inert"); d.append(body);
    $("#scrim").classList.add("open"); d.classList.add("open");
    overlayOpen(d,{label:"Ground truth for "+(tool||"this step"), onClose:closeDrawer});
  };
  if(!ev){ paint(liveEvidenceBody({missing:true, tool:tool})); return; }
  paint(liveEvidenceBody({payload:"", cycle:0, tool:tool, error:"Reading…"}));
  let dto;
  try{ dto = await liveGet("/v1/sessions/"+encodeURIComponent(ref)+"/evidence/"+encodeURIComponent(ev)); }
  catch(e){
    // 404 is the ORDINARY answer for a step recorded before capture existed — never an error banner.
    if(e && e.status === 404){ paint(liveEvidenceBody({missing:true, tool:tool})); return; }
    paint(liveEvidenceBody({tool:tool,
      error: e && e.status === 403
        ? "Reading recorded evidence needs an admin session — the same gate as the walk itself."
        : "The evidence read failed"+(e&&e.status?" ("+e.status+")":"")+". That is a read failure, not an empty observation."}));
    return;
  }
  paint(liveEvidenceBody(dto));
}

    old.replaceWith(btn);
  });

}

/* One render pass shares one clock origin, so every sample and every axis tick agree. */
let sigLiveNowMs = 0;

/* A series over REAL rows in the design's obs(t) contract: t is hours relative to now (negative = past).
   Bucket width tracks sigState.range, so one series renders correctly at 1h / 6h / 24h without re-fetching. */
function sigLiveSeries(rows, tsField){
  const ts = [];
  (Array.isArray(rows) ? rows : []).forEach(r => {
    const raw = r && (r[tsField] || r.created_at || r.opened_at || r.classified_at);
    const n = raw ? Date.parse(raw) : NaN;
    if (!isNaN(n)) ts.push(n);
  });
  ts.sort((a,b) => a-b);
  const fn = function(t){
    const R = (typeof sigRANGES !== "undefined" && sigRANGES[sigState.range]) || { hrs: 6 };
    const w = (R.hrs * 3600 * 1000) / 24;              // 24 buckets across the visible window
    const c = sigLiveNowMs + t * 3600 * 1000;
    let n = 0;
    for (let i = 0; i < ts.length; i++){ const x = ts[i]; if (x >= c - w/2 && x < c + w/2) n++; }
    return n;
  };
  fn.rowCount = ts.length;
  fn.stamps = ts;
  return fn;
}

/* The y domain comes from the DATA, never a fixed invented scale: sample the window, round up to clean ticks.
   A flat-zero series still gets a 0..4 axis so the panel reads as "measured zero", not as a broken chart. */
function sigLiveDomain(fn){
  const R = (typeof sigRANGES !== "undefined" && sigRANGES[sigState.range]) || { hrs: 6 };
  let mx = 0;
  for (let i = 0; i <= 24; i++){ const v = fn(-R.hrs + R.hrs * i / 24); if (v > mx) mx = v; }
  const top = mx <= 4 ? 4 : Math.ceil(mx / 4) * 4;
  return { dom: [0, top], yt: [0, top/4, top/2, 3*top/4, top].map(v => Math.round(v)) };
}

/* REAL TG events for the design's marker layer: deviations at the moment they were sealed. */
function sigLiveEvents(actions){
  const out = [];
  (Array.isArray(actions) ? actions : []).forEach(a => {
    if (!a || a.verdict !== "deviation") return;
    const n = a.sealed_at ? Date.parse(a.sealed_at) : NaN;
    if (isNaN(n)) return;
    out.push({ t: (n - sigLiveNowMs) / 3600000, glyph: "▲",
      label: "deviation " + String(a.action_id || a.external_ref || "").slice(0, 8), tone: "deviation" });
  });
  return out;
}

/* ===================== THE PER-MODULE CONFIGURATION DIALOG =====================

   The #modules drawer already existed as a DESIGN PREVIEW: every control present, every control
   disabled, "gated — design preview, no connector runtime". A complete form that could not save
   anything, which is the same defect class as a lane that is wired and never called — it looks like
   the feature exists.

   The form is GENERATED from /v1/modules/schema, never hand-written. TG has 42 connector packages;
   a hand-written dialog per module drifts from the binary the first time a module gains a field, and
   nothing compares the two. The module declares its fields; the console renders whatever it declares.

   THREE THINGS THIS SURFACE REFUSES TO DO:

     1. Claim a save took effect when it did not. Every field carries its declared EFFECT: a `live`
        field is re-read per use and lands immediately; a `restart` field is durable but inert until
        the worker restarts. The dialog says which, per field, before and after saving.
     2. Show a secret. The writer's OpenBao policy has no `read` capability, so the value cannot be
        fetched even by the process that wrote it. The field renders set/unset and rotates, never a
        value — and that is a property of the credential, not a UI choice.
     3. Imply completeness. Modules with no descriptor are listed by name, because a console showing
        only what it has a form for cannot distinguish "this is everything" from "this is what someone
        got round to describing".                                                                    */

/* liveModuleSchema fetches the generated-dialog shape. Cached per render pass; null = unavailable,
   which is rendered as unavailable rather than as an empty module list. */
async function modCfgLoadSchema(){
  try{
    const r = await fetch("/api/v1/modules/schema",{ credentials:"same-origin" });
    if(!r.ok) return null;
    return await r.json();
  }catch(e){ return null; }
}

/* modCfgEffect renders the honesty label for one field. This is the sentence that keeps the Save
   button from lying. */
function modCfgEffectNote(effect){
  if(effect === "live")
    return { text:"takes effect immediately — re-read on every use", color:"var(--match)" };
  if(effect === "restart")
    return { text:"saved durably, but the worker reads this at boot — it applies on the next restart",
             color:"var(--partial)" };
  return { text:"display only", color:"var(--ink4)" };
}

/* modCfgSaveField writes ONE non-secret field through the ledgered config route. */
async function modCfgSaveField(field, value, msgEl){
  if(!field.config_key){
    msgEl.style.color="var(--ink4)"; msgEl.textContent="this field has no config key — it is display only";
    return;
  }
  msgEl.style.color="var(--ink3)"; msgEl.textContent="recording…";
  let r=null;
  try{
    r = await fetch("/api/v1/config/"+encodeURIComponent(field.config_key),{
      method:"POST", credentials:"same-origin", headers:{ "Content-Type":"application/json" },
      body: JSON.stringify({ value:value, rationale:"module configuration dialog" }) });
  }catch(e){}
  if(r && r.ok){
    const out = await r.json().catch(()=>({}));
    const eff = modCfgEffectNote(field.effect);
    msgEl.style.color = eff.color;
    msgEl.textContent = "saved (ledger seq "+(out.ledger_seq!=null?out.ledger_seq:"?")+") — "+eff.text;
    return;
  }
  const f = r ? await cfgWriteFail(r) : { msg:"network error — retry" };
  msgEl.style.color="var(--deviation)"; msgEl.textContent=f.msg;
  if(f.elevate) cfgElevateOverlay("Changing a module's configuration needs an elevated session.");
}

/* modCfgSaveSecret sends the credential down its OWN lane — never the config store, which is
   Postgres-backed and ledgered. The input is cleared on success because the console has no way to
   read it back and must not appear to hold it. */
async function modCfgSaveSecret(mod, input, msgEl){
  const v = input.value;
  if(!v){ msgEl.style.color="var(--ink4)"; msgEl.textContent="enter the credential first"; return; }
  msgEl.style.color="var(--ink3)"; msgEl.textContent="storing…";
  let r=null;
  try{
    r = await fetch("/api/v1/modules/"+encodeURIComponent(mod.surface)+"/"+encodeURIComponent(mod.source_type)+"/secret",{
      method:"POST", credentials:"same-origin", headers:{ "Content-Type":"application/json" },
      body: JSON.stringify({ value:v }) });
  }catch(e){}
  if(r && r.ok){
    const out = await r.json().catch(()=>({}));
    input.value = "";   // the console cannot read it back; it must not look like it kept it
    msgEl.style.color="var(--match)";
    msgEl.textContent = "stored at "+(out.kv_path||"the secret backend")+"#"+(out.field||"?")
      + " — takes effect on the next call. It cannot be read back, here or anywhere.";
    return;
  }
  if(r && r.status===503){
    msgEl.style.color="var(--deviation)";
    msgEl.textContent="no secret write path on this control plane — the writer credential is not provisioned";
    return;
  }
  const f = r ? await cfgWriteFail(r) : { msg:"network error — retry" };
  msgEl.style.color="var(--deviation)"; msgEl.textContent=f.msg;
  if(f.elevate) cfgElevateOverlay("Storing a module credential needs an elevated session.");
}

/* modCfgTest presses the real module. The verb is shown BEFORE the press — "post a test message to
   the approvals room" is consent; an unlabelled TEST button is not. */
async function modCfgTest(mod, msgEl, btn){
  btn.disabled = true;
  msgEl.style.color="var(--ink3)"; msgEl.textContent="exercising the real module…";
  let r=null;
  try{
    r = await fetch("/api/v1/modules/"+encodeURIComponent(mod.surface)+"/"+encodeURIComponent(mod.source_type)+"/test",{
      method:"POST", credentials:"same-origin" });
  }catch(e){}
  btn.disabled = false;
  if(r && r.ok){
    const out = await r.json().catch(()=>({}));
    msgEl.style.color = out.ok ? "var(--match)" : "var(--deviation)";
    msgEl.textContent = (out.ok ? "PASS — " : "FAIL — ") + (out.summary||"")
      + (out.detail ? " · " + out.detail : "")
      + (out.elapsed_ms!=null ? " ("+out.elapsed_ms+"ms)" : "");
    return;
  }
  if(r && r.status===502){
    // The test could not be RUN, which is not the same as the module failing.
    msgEl.style.color="var(--deviation)";
    msgEl.textContent="the test could not be run — the worker did not answer. This is a TG-side problem, "
      + "not a verdict on the module.";
    return;
  }
  const f = r ? await cfgWriteFail(r) : { msg:"network error — retry" };
  msgEl.style.color="var(--deviation)"; msgEl.textContent=f.msg;
  if(f.elevate) cfgElevateOverlay("Testing a module needs an elevated session.");
}

/* modCfgField renders one generated input row. */
function modCfgField(mod, f, cfgValues){
  const row = h("div",{style:"padding:10px 0;border-top:1px solid var(--line2)"});
  const eff = modCfgEffectNote(f.effect);
  const head = h("div",{style:"display:flex;align-items:baseline;gap:8px;flex-wrap:wrap"},
    h("span",{style:"font-size:12px;color:var(--ink)"}, f.label || f.name),
    f.required ? h("span",{class:"lbl",style:"color:var(--partial)"},"required") : null,
    f.security==="authority" ? h("span",{class:"lbl",style:"color:var(--hazard)",
      title:"this field decides WHO is trusted — a change here moves the trust boundary"},"authority") : null,
    f.security==="secret" ? h("span",{class:"lbl",style:"color:var(--hazard)"},"secret") : null,
    h("span",{class:"mono",style:"font-size:10px;color:var(--ink4);margin-left:auto"}, f.env_key||""));
  row.append(head);
  if(f.help) row.append(h("div",{class:"desc",style:"font-size:11px;margin:2px 0 6px"}, f.help));

  const msg = h("div",{style:"font-size:11px;margin-top:6px;min-height:14px;color:"+eff.color}, eff.text);

  if(f.type === "secret_value"){
    // WRITE-ONLY. There is no value to prefill: the writer's policy grants no read, so the console
    // could not show one even if it wanted to.
    const input = h("input",{type:"password",class:"mono",placeholder:"paste the new credential",
      "aria-label":(f.label||f.name),
      style:"flex:1;background:var(--g3s);border:1px solid var(--line2);border-radius:var(--radius);padding:7px 10px;color:var(--ink);font-size:12px"});
    const btn = h("button",{class:"btn",type:"button",style:"font-size:10.5px;padding:4px 12px"},"STORE");
    btn.addEventListener("click", ()=>modCfgSaveSecret(mod, input, msg));
    row.append(h("div",{style:"display:flex;gap:8px;align-items:center"}, input, btn));
    msg.textContent = "write-only — stored in the secret backend and never read back into this dialog";
    msg.style.color = "var(--ink4)";
    row.append(msg);
    return row;
  }
  if(f.effect === "readonly"){
    row.append(h("div",{class:"mono",style:"font-size:11px;color:var(--ink3);padding:4px 0"},
      (cfgValues&&cfgValues[f.config_key]) || "(set outside this dialog)"));
    row.append(h("div",{class:"desc",style:"font-size:11px"},
      "display only — the operator sets the credential, not the pointer to it"));
    return row;
  }
  const input = h("input",{class:"mono","aria-label":(f.label||f.name),
    value:(cfgValues&&cfgValues[f.config_key])||"", placeholder:f.pattern||"",
    style:"flex:1;background:var(--g3s);border:1px solid var(--line2);border-radius:var(--radius);padding:7px 10px;color:var(--ink);font-size:12px"});
  const btn = h("button",{class:"btn",type:"button",style:"font-size:10.5px;padding:4px 12px"},"SAVE");
  btn.addEventListener("click", ()=>modCfgSaveField(f, input.value, msg));
  row.append(h("div",{style:"display:flex;gap:8px;align-items:center"}, input, btn));
  row.append(msg);
  return row;
}

/* modCfgCard renders one module's whole dialog. */
function modCfgCard(mod, cfgValues){
  const card = h("div",{class:"wrapcard",style:"margin-bottom:12px"});
  card.append(h("div",{class:"sec",style:"padding:10px 14px 4px"},
    h("span",{class:"lbl"}, mod.surface),
    h("h3",{style:"font-size:13px"}, mod.title || mod.source_type),
    // THREE STATES, NOT TWO. The API process registers ingest receivers and declares the model,
    // actuation and observability surfaces; the notifiers, trackers and the rest live in the worker,
    // which cannot publish its registry here. Rendering that absence as "not configured" told the
    // operator that a Matrix notifier delivering governance polls was switched off. Unknown must read as
    // unknown — it is the difference between a fact and the absence of one.
    (!mod.enabled_known
      ? h("span",{class:"lbl",style:"color:var(--partial)",title:"This module runs in the worker. The API process has no view of the worker's registry, so it cannot report state either way — see TG-251."},"state not reported here")
      : mod.enabled ? h("span",{class:"lbl",style:"color:var(--match)"},"enabled")
                    : h("span",{class:"lbl",style:"color:var(--ink4)"},"not configured")),
    h("span",{class:"rule"})));
  if(mod.summary) card.append(h("p",{class:"desc",style:"margin:0 14px 6px;font-size:11.5px"}, mod.summary));

  const body = h("div",{style:"padding:0 14px 12px"});
  (mod.fields||[]).forEach(f => body.append(modCfgField(mod, f, cfgValues)));

  // TEST — the verb is disclosed before the press.
  const testMsg = h("div",{style:"font-size:11px;margin-top:6px;min-height:14px;color:var(--ink4)"},
    mod.test_verb ? ("pressing TEST will "+mod.test_verb+" — it exercises the real module, not a mock")
                  : "no test is implemented for this module");
  const testBtn = h("button",{class:"btn",type:"button",style:"font-size:10.5px;padding:4px 14px"},"TEST");
  if(!mod.test_verb){ testBtn.disabled = true; testBtn.setAttribute("aria-disabled","true"); }
  testBtn.addEventListener("click", ()=>modCfgTest(mod, testMsg, testBtn));
  body.append(h("div",{style:"padding:12px 0 0;border-top:1px solid var(--line2);display:flex;gap:8px;align-items:center"},
    testBtn, h("span",{class:"desc",style:"font-size:11px"}, mod.test_verb||"")));
  body.append(testMsg);
  card.append(body);
  return card;
}
